local IntroSound = Instance.new("Sound")
IntroSound.Volume = 1.0
IntroSound.Parent = SoundService

task.spawn(function()
    local soundUrl = "https://files.catbox.moe/jkbi33.mp3"
    local fileName = "anomaly_intro_sound.mp3"
    
    if readfile and isfile and isfile(fileName) then
        if getasset then
            IntroSound.SoundId = getasset(fileName)
            IntroSound:Play()
            return
        end
    end

    local downloadSuccess, soundData = pcall(function()
        return game:HttpGet(soundUrl)
    end)
    
    if downloadSuccess and soundData then
        if writefile then writefile(fileName, soundData) end
        if getasset then
            IntroSound.SoundId = getasset(fileName)
            IntroSound:Play()
        end
    end
end)

    TxtBox:GetPropertyChangedSignal("Text"):Connect(function()
        local cleaned = TxtBox.Text:gsub("[^0-9%.%-]", "")
        -- birden fazla nokta engelle
        local firstDot = cleaned:find("%.")
        if firstDot then
            cleaned = cleaned:sub(1, firstDot) .. cleaned:sub(firstDot + 1):gsub("%.", "")
        end
        if TxtBox.Text ~= cleaned then
            TxtBox.Text = cleaned
        end
    end)

    BoxCorner.CornerRadius = UDim.new(0, 4)
    BoxCorner.Parent = TxtBox

    BoxStroke.Transparency = 0.4

        return CFrame.new(0, 10, 0)

    local function findSolidBelow(char, hrp)
        local exclude = { char }
        local origin = hrp.Position
        local params = RaycastParams.new()
        params.FilterType = Enum.RaycastFilterType.Exclude
        params.FilterDescendantsInstances = exclude
        params.IgnoreWater = true

        local from = origin
        for _ = 1, 25 do
            local result = workspace:Raycast(from, Vector3.new(0, -800, 0), params)
            if not result then return nil end
            local inst = result.Instance
            -- görünmez / collidesiz blokları atla
            local vis = true
            pcall(function()
                if inst.Transparency >= 0.95 then vis = false end
                if inst.LocalTransparencyModifier and inst.LocalTransparencyModifier >= 0.95 then vis = false end
                if not inst.CanCollide then vis = false end
            end)
            if vis then
                return result.Position + Vector3.new(0, 3.2, 0)
            end
            table.insert(exclude, inst)
            params.FilterDescendantsInstances = exclude
            from = result.Position - Vector3.new(0, 0.15, 0)
        end
        return nil
    end

        local char = LP.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end

        local method = (getTpDownMethod and getTpDownMethod()) or "Normal"

        if method == "Normal" then

        -- Slow Fall / Zigzag: alttan inerken WASD ile hareket

        tpDownActive = true
        AnomalyTpDownActive = true
        zigPhase = 0
        AnomalyHooksOff = false
        AnomalyForceVel = nil -- ForceVel kilitlemesin; doğrudan yazacağız
        local startY = hrp.Position.Y
        local groundY = nil
        local gpos = findSolidBelow(char, hrp)
        if gpos then groundY = gpos.Y end

        local function getHorizontalMove(moveSpd)
            -- freefall'da MoveDirection 0 olabilir → kamera + tuş
            local cam = workspace.CurrentCamera
            local move = Vector3.zero
            if cam then
                local look = cam.CFrame.LookVector
                local right = cam.CFrame.RightVector
                look = Vector3.new(look.X, 0, look.Z)
                right = Vector3.new(right.X, 0, right.Z)
                if look.Magnitude > 0.01 then look = look.Unit end
                if right.Magnitude > 0.01 then right = right.Unit end
                if UserInputService:IsKeyDown(Enum.KeyCode.W) then move = move + look end
                if UserInputService:IsKeyDown(Enum.KeyCode.S) then move = move - look end
                if UserInputService:IsKeyDown(Enum.KeyCode.A) then move = move - right end
                if UserInputService:IsKeyDown(Enum.KeyCode.D) then move = move + right end
            end
            -- mobile / gamepad MoveDirection yedek
            local c = LP.Character
            local hum = c and c:FindFirstChildOfClass("Humanoid")
            if move.Magnitude < 0.05 and hum and hum.MoveDirection.Magnitude > 0.05 then

            end
            if move.Magnitude > 0.05 then
                return move.Unit * moveSpd
            end
            return Vector3.zero
        end

                return

                AnomalyTpDownActive = false
                return

                local zz = math.sin(zigPhase * 0.5) * 28

            AnomalyHooksOff = false

                pcall(function()
                    r.CFrame = r.CFrame + Vector3.new(horiz.X, 0, horiz.Z) * dt
                end)
            end
            -- dikey yavaş düşüş
            AnomalyVelWrite = true
            pcall(function()
                local v = r.AssemblyLinearVelocity
                r.AssemblyLinearVelocity = Vector3.new(v.X * 0.2 + horiz.X * 0.5, vy, v.Z * 0.2 + horiz.Z * 0.5)
            end)
            AnomalyVelWrite = false
        end)
    end

        end

local MultiJumpPowerCard, MultiJumpPowerTextBox = createOptionCard(22, "Multi Jump Power", "45")

-- Multi Jump engine: AssemblyLinearVelocity
do
    local multiJumpOn = false
    local lastHoldJump = 0
    local spaceHeld = false
    local clickUsedThisHold = false

    MultiJumpToggle.MouseButton1Click:Connect(function()
        multiJumpOn = not multiJumpOn
    end)

    local function doMultiJump()
        local char = game:GetService("Players").LocalPlayer.Character
        local hrp = char and char:FindFirstChild("HumanoidRootPart")
        if not hrp then return end
        local power = tonumber(MultiJumpPowerTextBox and MultiJumpPowerTextBox.Text) or 45
        if power < 1 then power = 45 end
        local vx = hrp.AssemblyLinearVelocity.X
        local vz = hrp.AssemblyLinearVelocity.Z
        AnomalyVelWrite = true
        pcall(function()
            hrp.AssemblyLinearVelocity = Vector3.new(vx, power, vz)
        end)
        AnomalyVelWrite = false
    end

    -- Click: sadece basışta 1 kez (bırakıp yeniden basana kadar tekrar yok)
    -- Hold: basışta + basılı tutunca 0.3s aralık
    UserInputService.InputBegan:Connect(function(input, gp)
        if gp or not multiJumpOn then return end
        if input.KeyCode == Enum.KeyCode.Space or input.KeyCode == Enum.KeyCode.ButtonA then
            if selectedMultiJumpMode == "Click" then
                if not clickUsedThisHold then
                    doMultiJump()
                    clickUsedThisHold = true
                end
            else
                doMultiJump()
                lastHoldJump = tick()
            end
            spaceHeld = true
        end
    end)

    UserInputService.InputEnded:Connect(function(input)
        if input.KeyCode == Enum.KeyCode.Space or input.KeyCode == Enum.KeyCode.ButtonA then
            spaceHeld = false
            clickUsedThisHold = false
        end
    end)

    -- JumpRequest: Click modda yok say (tutunca spam olmasın)
    pcall(function()
        UserInputService.JumpRequest:Connect(function()
            if not multiJumpOn then return end
            if selectedMultiJumpMode == "Click" then return end
            -- Hold: JumpRequest ile de destekle
            if tick() - lastHoldJump >= 0.3 then
                lastHoldJump = tick()
                doMultiJump()
            end
        end)
    end)

    RunService.Heartbeat:Connect(function()
        if not multiJumpOn then return end
        if selectedMultiJumpMode ~= "Hold" then return end
        local held = spaceHeld or UserInputService:IsKeyDown(Enum.KeyCode.Space)
        if not held then return end
        if tick() - lastHoldJump < 0.3 then return end
        lastHoldJump = tick()
        doMultiJump()
    end)
end

-- Shared HUD state (Anomaly Hub Auto Steal UI bağlanır)
AnomalyStealState = {
    mode = "Normal",
    progress = 0,
    status = "", -- Semi: Unready/Ready; Normal/Fast: boş

-- ========== AUTO STEAL ENGINE ==========
do
    local PlayersSvc = game:GetService("Players")
    local LP = PlayersSvc.LocalPlayer
    local autoStealOn = false
    local autoStealBusy = false
    local autoStealConn = nil
    local promptCache = {} -- [prompt] = {hold={}, trigger={}, ready=true}

    local function hudSet(fields)
        for k, v in pairs(fields) do
            AnomalyStealState[k] = v
        end
    end

    local function getRadius()
        local n = tonumber(AutoStealRadiusBox and AutoStealRadiusBox.Text)
        if not n or n <= 0 then n = 60 end
        return n
    end

    local function getHRP()
        local char = LP and LP.Character
        if not char then return nil end
        return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("UpperTorso") or char.PrimaryPart
    end

    local function isStealPrompt(prompt)
        if not prompt or not prompt:IsA("ProximityPrompt") then return false end
        local a = (prompt.ActionText or ""):lower()
        local o = (prompt.ObjectText or ""):lower()
        local n = (prompt.Name or ""):lower()
        return a:find("steal", 1, true) or o:find("steal", 1, true) or n:find("steal", 1, true)
    end

    local function promptPosition(prompt)
        local parent = prompt.Parent
        if parent and parent:IsA("Attachment") then
            return parent.WorldPosition
        elseif parent and parent:IsA("BasePart") then
            return parent.Position
        elseif parent then
            local p = parent:FindFirstChildWhichIsA("BasePart", true)
            if p then return p.Position end
        end
        return nil
    end

    local function distToPrompt(prompt)
        local hrp = getHRP()
        local pos = promptPosition(prompt)
        if not hrp or not pos then return math.huge end
        return (hrp.Position - pos).Magnitude
    end

    -- ProximityPrompt cache (her seferinde GetDescendants YOK → lag düşer)
    local stealPromptList = {}
    local function addStealPrompt(obj)
        if obj:IsA("ProximityPrompt") and isStealPrompt(obj) then
            stealPromptList[obj] = true
        end
    end
    local function scanStealPrompts()
        local found = {}
        for prompt in pairs(stealPromptList) do
            if prompt.Parent and prompt:IsA("ProximityPrompt") and prompt.Enabled and isStealPrompt(prompt) then
                table.insert(found, prompt)
            else
                stealPromptList[prompt] = nil
            end
        end
        return found
    end
    task.defer(function()
        for _, obj in ipairs(workspace:GetDescendants()) do
            addStealPrompt(obj)
        end
        workspace.DescendantAdded:Connect(function(obj)
            addStealPrompt(obj)
        end)
        workspace.DescendantRemoving:Connect(function(obj)
            if stealPromptList[obj] then stealPromptList[obj] = nil end
        end)
    end)

    local function cachePromptCallbacks(prompt)
        if promptCache[prompt] then return promptCache[prompt] end
        local data = { hold = {}, trigger = {}, ready = true }
        pcall(function()
            if getconnections then
                local ok, conns = pcall(getconnections, prompt.PromptButtonHoldBegan)
                if ok and type(conns) == "table" then
                    for _, c in ipairs(conns) do
                        if type(c.Function) == "function" then
                            table.insert(data.hold, c.Function)
                        end
                    end
                end
                local ok2, conns2 = pcall(getconnections, prompt.Triggered)
                if ok2 and type(conns2) == "table" then
                    for _, c in ipairs(conns2) do
                        if type(c.Function) == "function" then
                            table.insert(data.trigger, c.Function)
                        end
                    end
                end
            end
        end)
        promptCache[prompt] = data
        return data
    end

    local function fireHold(prompt)
        local data = cachePromptCallbacks(prompt)
        for _, fn in ipairs(data.hold) do
            task.spawn(fn)
        end
        pcall(function()
            if fireproximityprompt then
                -- bazı executorlarda hold için
            end
        end)
    end

    local function fireTrigger(prompt)
        local data = cachePromptCallbacks(prompt)
        for _, fn in ipairs(data.trigger) do
            task.spawn(fn)
        end
        pcall(function()
            if fireproximityprompt then
                fireproximityprompt(prompt)
            end
        end)
    end

    -- ----- Normal / Fast loop -----
    local function runNormalOrFast(mode)
        if autoStealBusy then return end
        local radius = getRadius()
        local prompts = scanStealPrompts()
        local nearest, nearestDist = nil, math.huge
        for _, p in ipairs(prompts) do
            local d = distToPrompt(p)
            if d <= radius and d < nearestDist then
                nearest, nearestDist = p, d
            end
        end
        if not nearest then return end

        local data = cachePromptCallbacks(nearest)
        if not data.ready then return end
        data.ready = false
        autoStealBusy = true

        task.spawn(function()
            if mode == "Fast" then
                pcall(function() nearest.HoldDuration = 1.3 end)
            end
            local holdTime = nearest.HoldDuration
            if not holdTime or holdTime <= 0 then holdTime = (mode == "Fast") and 1.3 or 0.5 end
            if mode == "Fast" then holdTime = 1.3 end

            local targetLabel = nearest.ObjectText ~= "" and nearest.ObjectText or (nearest.ActionText ~= "" and nearest.ActionText or "Steal")
            hudSet({
                holding = true,
                progress = 0,
                status = "", -- Normal/Fast: ready/unready yok
                targetName = targetLabel,
                mode = mode,
            })

            fireHold(nearest)
            local start = tick()
            while tick() - start < holdTime and autoStealOn and nearest.Parent do
                if distToPrompt(nearest) > radius then break end
                local ratio = math.clamp((tick() - start) / holdTime, 0, 1)
                hudSet({ progress = ratio })
                task.wait()
            end
            if autoStealOn and nearest.Parent and distToPrompt(nearest) <= radius then
                fireTrigger(nearest)
                hudSet({ progress = 1 })
            end
            data.ready = true
            autoStealBusy = false
            hudSet({ holding = false, progress = 0, targetName = "None" })
        end)
    end

    -- ----- Semi (Casual Dragon logic) -----
    local semiState = {
        caches = {},
        connections = {},
        animals = {},
        promptMap = {},
        active = false,
    }

    local function semiPath(path, root)
        if typeof(path) == "table" then
            local cur = root
            for _, k in ipairs(path) do
                if not cur then return nil end
                cur = cur[k]
            end
            return cur
        end
        local parts = {}
        for m in string.gmatch(tostring(path), "[^%.]+") do
            table.insert(parts, tonumber(m) or m)
        end
        local cur = root
        for _, k in ipairs(parts) do
            if not cur then return nil end
            cur = cur[k]
        end
        return cur
    end

    local function semiApplyPatch(plotName, patch)
        local cache = semiState.caches[plotName]
        if typeof(cache) ~= "table" then return end
        local path, op, val, key = patch[1], patch[2], patch[3], patch[4]
        local node, parent, lastKey
        local cur = cache
        local keys = {}
        if typeof(path) == "table" then
            keys = path
        else
            for m in string.gmatch(tostring(path), "[^%.]+") do
                table.insert(keys, tonumber(m) or m)
            end
        end
        for i, k in ipairs(keys) do
            parent, lastKey = cur, k
            if cur then cur = cur[k] end
            if i == #keys then break end
        end
        if op == "Changed" and parent and lastKey ~= nil then
            parent[lastKey] = val
        elseif op == "ArrayInsert" and cur then
            table.insert(cur, key, val)
        elseif op == "ArrayRemoved" and cur then
            table.remove(cur, key)
        elseif op == "DictionaryInsert" and cur then
            cur[key] = val
        elseif op == "DictionaryRemoved" and cur then
            cur[key] = nil
        end
    end

    local function semiHookChannel(remote)
        if not remote or semiState.connections[remote] then return end
        local plotName = tostring(remote.Name)
        local plots = workspace:FindFirstChild("Plots")
        if not plots or not plots:FindFirstChild(plotName) then return end

        local rs = game:GetService("ReplicatedStorage")
        local packages = rs:FindFirstChild("Packages")
        local sync = packages and packages:FindFirstChild("Synchronizer")
        local requestData = sync and sync:FindFirstChild("RequestData")

        if requestData and semiState.caches[plotName] == nil then
            local ok, result = pcall(function()
                return requestData:InvokeServer(plotName)
            end)
            if ok and typeof(result) == "table" then
                semiState.caches[plotName] = result
            else
                semiState.caches[plotName] = {}
            end
        elseif semiState.caches[plotName] == nil then
            semiState.caches[plotName] = {}
        end

        semiState.connections[remote] = remote.OnClientEvent:Connect(function(list)
            for _, patch in ipairs(list) do
                semiApplyPatch(plotName, patch)
            end
        end)
    end

    local function semiInitChannels()
        local rs = game:GetService("ReplicatedStorage")
        local packages = rs:FindFirstChild("Packages")
        local sync = packages and packages:FindFirstChild("Synchronizer")
        if not sync then return end
        local channel = sync:FindFirstChild("Channel")
        if not channel then return end
        for _, child in ipairs(channel:GetChildren()) do
            if child:IsA("RemoteEvent") then
                pcall(semiHookChannel, child)
            end
        end
        channel.ChildAdded:Connect(function(child)
            if child:IsA("RemoteEvent") then
                pcall(semiHookChannel, child)
            end
        end)
        local route = sync:FindFirstChild("CommunicationRoute")
        if route and route:IsA("RemoteEvent") then
            route.OnClientEvent:Connect(function(list)
                for _, v in ipairs(list) do
                    local op, name = v[1], tostring(v[2])
                    if op == "ListenerAdded" then
                        local ev = channel:FindFirstChild(name)
                        if ev and ev:IsA("RemoteEvent") then pcall(semiHookChannel, ev) end
                    end
                end
            end)
        end
    end

    local function semiRefreshAnimals()

        local plots = workspace:FindFirstChild("Plots")
        if not plots then return 0 end
        local animalsData
        pcall(function()
            animalsData = require(game:GetService("ReplicatedStorage"):WaitForChild("Datas"):WaitForChild("Animals"))
        end)
        for _, plot in ipairs(plots:GetChildren()) do
            local cache = semiState.caches[plot.Name]
            if cache and typeof(cache.AnimalList) == "table" then
                for slot, entry in pairs(cache.AnimalList) do
                    if type(entry) == "table" and entry.Index then
                        local info = animalsData and animalsData[entry.Index]

                            name = display,
                            plot = plot.Name,
                            slot = tostring(slot),
                            uid = plot.Name .. "_" .. tostring(slot),

                end

        end

        return #list
    end

    local function semiNearest()

    local HOLD_MIN, HOLD_MAX, ENTRY_DELAY, STEAL_RANGE = 1.3, 2.6, 0.3, 10

    local function semiTrySteal(prompt, animal)
        if semiState.active then return false end
        local data = cachePromptCallbacks(prompt)
        if not data.ready then return false end
        data.ready = false
        semiState.active = true
        autoStealBusy = true

        task.spawn(function()
            local start = tick()
            hudSet({
                holding = true,
                progress = 0,
                status = "Unready",
                targetName = animal.name or "Animal",
                mode = "Semi",
            })
            for _, fn in ipairs(data.hold) do task.spawn(fn) end

            -- 1.3 sn hold: progress 0 -> ~0.5, sonra Ready
            while tick() - start < HOLD_MIN and autoStealOn and prompt.Parent do
                local ratio = math.clamp((tick() - start) / HOLD_MAX, 0, 1)
                hudSet({ progress = ratio, status = "Unready" })
                task.wait()
            end
            if autoStealOn then
                hudSet({ status = "Ready" })
            end

            local inRange = false
            local hrp = getHRP()
            local pos = semiSlotPos(animal)
            if hrp and pos then
                inRange = (hrp.Position - pos).Magnitude <= STEAL_RANGE
            end
            local success = false
            while tick() - start < HOLD_MAX and prompt.Parent and autoStealOn do
                local ratio = math.clamp((tick() - start) / HOLD_MAX, 0, 1)
                hudSet({ progress = ratio, status = "Ready" })
                hrp = getHRP()
                pos = semiSlotPos(animal)
                local d = (hrp and pos) and (hrp.Position - pos).Magnitude or math.huge
                if d <= STEAL_RANGE then
                    if not inRange then task.wait(ENTRY_DELAY) end
                    for _, fn in ipairs(data.trigger) do task.spawn(fn) end
                    success = true
                    hudSet({ progress = 1 })
                    break
                end
                task.wait()
            end
            semiState.active = false
            autoStealBusy = false
            hudSet({ holding = false, progress = 0, status = "", targetName = "None" })
            task.wait(0.05)
            data.ready = true
        end)
        return true
    end

    local semiInited = false
    local lastAnimalRefresh = 0

    local function runSemi()
        if not semiInited then
            pcall(semiInitChannels)
            semiInited = true
        end
        if tick() - lastAnimalRefresh > 5 then
            pcall(semiRefreshAnimals)
            lastAnimalRefresh = tick()
        end
        if semiState.active or autoStealBusy then return end
        local target = semiNearest()
        if not target then return end
        local prompt = semiFindPrompt(target)
        if prompt then
            semiTrySteal(prompt, target)
        end
    end

    -- ana döngü
    local function startAutoSteal()
        if autoStealConn then return end
        hudSet({ enabled = true })
        autoStealConn = task.spawn(function()
            while autoStealOn do
                local mode = getAutoStealMethod and getAutoStealMethod() or "Normal"
                hudSet({ mode = mode })
                if not AnomalyStealState.holding then
                    -- idle: Normal/Fast status boş, Semi boş
                    if mode ~= "Semi" then
                        hudSet({ status = "" })
                    end
                end
                pcall(function()
                    if mode == "Semi" then
                        runSemi()
                    else
                        runNormalOrFast(mode)
                    end
                end)
                task.wait(0.2)
            end
            autoStealConn = nil
        end)
    end

    local function stopAutoSteal()
        autoStealOn = false
        autoStealBusy = false
        semiState.active = false
        hudSet({
            enabled = false,
            holding = false,
            progress = 0,
            status = "",
            targetName = "None",
        })
    end

    AutoStealToggleBtn.MouseButton1Click:Connect(function()
        autoStealOn = not autoStealOn
        if autoStealOn then
            startAutoSteal()
        else
            stopAutoSteal()
        end
    end)
end

    local function getBatSpeed()
        local n = tonumber(BatAimbotSpeedTextBox and BatAimbotSpeedTextBox.Text)
        if not n or n <= 0 then n = 65 end
        return n
    end

    local function findBatTool(char)
        if not char then return nil end
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("bat", 1, true) then
                return t
            end
        end
        local backpack = LP:FindFirstChild("Backpack")
        if backpack then
            for _, t in ipairs(backpack:GetChildren()) do
                if t:IsA("Tool") and t.Name:lower():find("bat", 1, true) then
                    return t
                end
            end
        end
        return nil
    end

    local function equipBat()
        local char = LP.Character
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if not hum then return end
        -- zaten eldeyse tamam
        for _, t in ipairs(char:GetChildren()) do
            if t:IsA("Tool") and t.Name:lower():find("bat", 1, true) then
                return
            end
        end
        local tool = findBatTool(char)
        if tool then
            pcall(function()
                hum:EquipTool(tool)
            end)
        end
    end

    local function nearestEnemyHRP()
        local char = LP.Character
        local myHRP = char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart)
        if not myHRP then return nil end
        local best, bestD = nil, math.huge
        for _, plr in ipairs(PlayersSvc:GetPlayers()) do
            if plr ~= LP and plr.Character then
                local hrp = plr.Character:FindFirstChild("HumanoidRootPart") or plr.Character.PrimaryPart
                local hum = plr.Character:FindFirstChildOfClass("Humanoid")
                if hrp and hum and hum.Health > 0 then
                    local d = (hrp.Position - myHRP.Position).Magnitude
                    if d < bestD then
                        best, bestD = hrp, d
                    end
                end
            end
        end
        return best
    end

    -- sideStates sonradan set edilir; global bayrak
    AnomalyBatAimbotOn = false

    -- Normal + V1: AssemblyLinearVelocity ile en yakına git
    -- Normal: hook kapalı | Bypass: hook açık
    AnomalyBatMode = "Normal"
    AnomalyBatVersion = "V1"

    RunService.Heartbeat:Connect(function()
        if not AnomalyBatAimbotOn then
            -- bat kapalıyken Normal modun hook kapatması kalsın diye zorlama yok
            return
        end
        -- V1 (Normal veya Bypass)
        if AnomalyBatVersion ~= "V1" and selectedVersion ~= "V1" then return end
        local mode = AnomalyBatMode or selectedMode or "Normal"
        local ver = AnomalyBatVersion or selectedVersion or "V1"
        if ver ~= "V1" then return end

        if AnomalyFling and AnomalyFling.active then return end

        local char = LP.Character
        local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart)
        if not hrp then return end

        equipBat()

        local target = nearestEnemyHRP()
        if not target then return end

        local speed = getBatSpeed()
        local dir = target.Position - hrp.Position
        local flat = Vector3.new(dir.X, 0, dir.Z)
        local y = hrp.AssemblyLinearVelocity.Y

        if mode == "Normal" then
            AnomalyHooksOff = true
        else
            -- Bypass: hook'lar devrede kalsın
            AnomalyHooksOff = false
        end

        AnomalyVelWrite = true
        if flat.Magnitude < 0.5 then
            pcall(function()
                hrp.AssemblyLinearVelocity = Vector3.new(0, y, 0)
            end)
            AnomalyVelWrite = false
            return
        end
        local unit = flat.Unit
        pcall(function()
            hrp.AssemblyLinearVelocity = Vector3.new(unit.X * speed, y, unit.Z * speed)
        end)
        AnomalyVelWrite = false
        pcall(function()
            hrp.CFrame = CFrame.new(hrp.Position, Vector3.new(target.Position.X, hrp.Position.Y, target.Position.Z))
        end)
    end)

        for _, c in ipairs(deathConns) do
            pcall(function() c:Disconnect() end)
        end
        deathConns = {}
        if heartConn then heartConn:Disconnect() heartConn = nil end
    end

    local function protectChar(char)
        if not char then return end
        local hum = char:FindFirstChildOfClass("Humanoid")
            or char:WaitForChild("Humanoid", 3)
        if not hum then return end

        hum.MaxHealth = math.huge
        hum.Health = math.huge
        hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)

        local sc = hum.StateChanged:Connect(function(_, new)
            if not shouldProtect() then return end
            if new == Enum.HumanoidStateType.Dead then
                hum.Health = math.huge
                hum:SetStateEnabled(Enum.HumanoidStateType.Dead, false)
            end
        end)
        table.insert(deathConns, sc)

        local hc = hum:GetPropertyChangedSignal("Health"):Connect(function()
            if not shouldProtect() then return end
            if hum.Health < hum.MaxHealth then
                hum.Health = math.huge
            end
        end)
        table.insert(deathConns, hc)

        if heartConn then heartConn:Disconnect() end
        heartConn = RunService.Heartbeat:Connect(function()
            if not shouldProtect() then return end
            if hum and hum.Parent and hum.Health < hum.MaxHealth then
                hum.Health = math.huge
            end
        end)
    end

    local function startProtect()
        clearDeathConns()
        if charAddedConn then charAddedConn:Disconnect() charAddedConn = nil end
        if not AnomalyAntiDie.on then return end
        protectChar(LP.Character)
        charAddedConn = LP.CharacterAdded:Connect(function(c)
            if not AnomalyAntiDie.on then return end
            task.wait(0.1)
            clearDeathConns()
            protectChar(c)
        end)
    end

    local function stopProtect()
        AnomalyAntiDie.forceUntil = 0
        clearDeathConns()
        if charAddedConn then charAddedConn:Disconnect() charAddedConn = nil end
        local char = LP.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                pcall(function()
                    hum:SetStateEnabled(Enum.HumanoidStateType.Dead, true)
                    hum.MaxHealth = 100
                    if hum.Health > 100 then hum.Health = 100 end
                end)
            end
        end
    end

    function AnomalyForceAntiDieUntil(_t)
        -- Drop Br artık zorla anti-die açmaz
    end

    AntiDieToggle.MouseButton1Click:Connect(function()
        AnomalyAntiDie.on = not AnomalyAntiDie.on
        if AnomalyAntiDie.on then
            startProtect()
        else
            stopProtect()
        end
    end)

    -- başlangıçta koruma kapalı
    AnomalyAntiDie.on = false
end

    local LP = game:GetService("Players").LocalPlayer
    local antiVoidOn = false
    local antiVoidConn = nil
    local lastTpTime = 0
    local DIST_LIMIT = 500
    local TP_INTERVAL = 0.6
    AnomalyInstantResetActive = AnomalyInstantResetActive or false

    local function getSpawnPos()
        local spawnLoc = workspace:FindFirstChildOfClass("SpawnLocation")
        if spawnLoc then
            return spawnLoc.Position + Vector3.new(0, 3, 0)
        end
        return Vector3.new(0, 10, 0)
    end

    local function setAntiVoid(on)
        antiVoidOn = on == true
        lastTpTime = 0
        if antiVoidConn then
            antiVoidConn:Disconnect()
            antiVoidConn = nil
        end
        if not antiVoidOn then return end

        antiVoidConn = RunService.Heartbeat:Connect(function()
            if not antiVoidOn then return end
            -- Instant Reset aktifken (respawn olana kadar) Anti Void çalışmasın
            if AnomalyInstantResetActive then return end
            local char = LP.Character
            local hrp = char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart)
            if not hrp then return end
            local spawnPos = getSpawnPos()
            local pos = hrp.Position
            local dist = (pos - spawnPos).Magnitude
            if dist <= DIST_LIMIT then
                return
            end
            local now = tick()
            if (now - lastTpTime) < TP_INTERVAL then return end
            lastTpTime = now
            pcall(function()
                hrp.CFrame = CFrame.new(spawnPos)
                hrp.AssemblyLinearVelocity = Vector3.zero
                hrp.AssemblyAngularVelocity = Vector3.zero
            end)
        end)
    end

    AntiVoidToggle.MouseButton1Click:Connect(function()
        setAntiVoid(not antiVoidOn)
    end)
end

    local function resolveRemote()
        local rrs = game:FindFirstChild("RobloxReplicatedStorage")
        if not rrs then return nil end
        local r = rrs:FindFirstChild("SetPlayerBlockList")
        if r and (r:IsA("RemoteEvent") or r:IsA("UnreliableRemoteEvent")) then
            return r
        end
        return nil
    end

    local function buildPayload()
        local root = table.create(Children)
        for i = 1, Children do
            local entry = { id = i, v = i * 2 }
            for k = 1, FieldsPerChild do
                entry["k" .. k] = (FieldSize > 0) and antiLoseState.pad or (i + k)
            end
            for j = 1, NestedPerChild do
                entry["n" .. j] = { a = i, b = j }
            end
            root[i] = entry
        end
        return root
    end

    local function fireAntiLose()
        if not antiLoseState.remote then
            antiLoseState.remote = resolveRemote()
            if not antiLoseState.remote then return end
        end
        local payload
        if RebuildEach or not antiLoseState.cache then
            payload = buildPayload()
            if not RebuildEach then antiLoseState.cache = payload end
        else
            payload = antiLoseState.cache
        end
        for _ = 1, FireBurst do
            pcall(function()
                antiLoseState.remote:FireServer(payload)
            end)
        end
    end

                    if not (hrp and humanoid) then return end

                    -- kamera ELLENMEZ; respawn'a kadar Anti Void kapalı
                    AnomalyInstantResetActive = true
                    AnomalyFling.active = true
                    AnomalyFling.kind = "InstantReset"
                    AnomalyForceVel = Vector3.new(0, 3e6, 0)

                        AnomalyFling.active = false
                        AnomalyFling.kind = nil
                        AnomalyForceVel = nil
                    end

                    local respawnConn
                    respawnConn = LP.CharacterAdded:Connect(function()
                        AnomalyInstantResetActive = false
                        stopAnyFling()
                        if respawnConn then respawnConn:Disconnect() end
                    end)

                elseif name == "DropBr" then
                    -- aynı anda 1 kez; açıkken tekrar bas → hiçbir şey
                    if AnomalyFling.active and AnomalyFling.kind == "DropBr" then
                        return
                    end
                    if AnomalyFling.active then
                        stopAnyFling()

                    local character = LP.Character
                    if not character then return end
                    if not character:FindFirstChild("HumanoidRootPart") then return end

                    AnomalyFling.kind = "DropBr"
                    local startT = tick()
                    local movel = 0.1
                    local TOTAL_END = 0.12

                        if not (AnomalyFling.active and AnomalyFling.kind == "DropBr") then return end

                        AnomalyFling.kind = nil

                        AnomalyHooksOff = false
                        AnomalyVelWrite = false

                                    end)

                                end

                            task.wait(0.02)

                    end)
                end

                elseif (name == "LaggerNormal" or name == "LaggerSteal") and newState then

                styleBtn(btn, newState)

                if name == "BatAimbot" then

                    if not newState then
                        AnomalyHooksOff = false
                    end
                elseif name == "StealSpeed" then
                    AnomalySpeedFlags.stealSpeed = newState
                    if newState then
                        AnomalySpeedFlags.laggerNormal = false
                        AnomalySpeedFlags.laggerSteal = false
                    end
                elseif name == "LaggerNormal" then

-- Build simple white R6-style dummy in a WorldModel (for ViewportFrame)
local function createWhiteDummy()
    local model = Instance.new("Model")
    model.Name = "Dummy"

    local function makePart(name, size, cf, color)
        local p = Instance.new("Part")
        p.Name = name
        p.Size = size
        p.CFrame = cf
        p.Anchored = true
        p.CanCollide = false
        p.Material = Enum.Material.SmoothPlastic
        p.Color = color or Color3.fromRGB(245, 245, 250)
        p.Parent = model
        return p
    end

    local torso = makePart("Torso", Vector3.new(2, 2, 1), CFrame.new(0, 1, 0))
    local head = makePart("Head", Vector3.new(1.2, 1.2, 1.2), CFrame.new(0, 2.6, 0))
    local headMesh = Instance.new("SpecialMesh")
    headMesh.MeshType = Enum.MeshType.Head
    headMesh.Scale = Vector3.new(1.25, 1.25, 1.25)
    headMesh.Parent = head

    -- smile face (Decal)
    local face = Instance.new("Decal")
    face.Face = Enum.NormalId.Front
    face.Texture = "rbxasset://textures/face.png" -- default smile-ish; fallback white head looks fine
    face.Parent = head

    makePart("Left Arm", Vector3.new(1, 2, 1), CFrame.new(-1.5, 1, 0))
    makePart("Right Arm", Vector3.new(1, 2, 1), CFrame.new(1.5, 1, 0))
    makePart("Left Leg", Vector3.new(1, 2, 1), CFrame.new(-0.5, -1, 0))
    makePart("Right Leg", Vector3.new(1, 2, 1), CFrame.new(0.5, -1, 0))

    -- slight left turn already applied via camera / model CFrame later
    model.PrimaryPart = torso
    return model
end

local function setupViewport(viewport, usePlayerAvatar)
    local world = Instance.new("WorldModel")
    world.Parent = viewport

    local cam = Instance.new("Camera")
    cam.Parent = viewport
    viewport.CurrentCamera = cam

    local model
    if usePlayerAvatar then
        local ok, result = pcall(function()
            return Players:CreateHumanoidModelFromUserId(localPlayer.UserId)
        end)
        if ok and result then
            model = result
            model.Parent = world
            -- remove scripts
            for _, d in ipairs(model:GetDescendants()) do
                if d:IsA("Script") or d:IsA("LocalScript") then
                    d:Destroy()
                end
            end
            local hum = model:FindFirstChildOfClass("Humanoid")
            if hum then hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end
        end
    end

    if not model then
        model = createWhiteDummy()
        model.Parent = world
    end

    -- position & slight left yaw
    local primary = model.PrimaryPart or model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso") or model:FindFirstChildWhichIsA("BasePart")
    if primary then
        -- yüz kameraya baksın, hafif sola dönük
        model:PivotTo(CFrame.new(0, 0, 0) * CFrame.Angles(0, math.rad(155), 0))
        local bbox = model:GetExtentsSize()
        local dist = math.max(bbox.X, bbox.Y, bbox.Z) * 1.35
        cam.CFrame = CFrame.new(Vector3.new(-0.35, bbox.Y * 0.35, dist), Vector3.new(0, bbox.Y * 0.28, 0))
    else
        cam.CFrame = CFrame.new(0, 1.5, 5)
    end
end

local function isPlayerCharacter(inst)
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr.Character and (inst == plr.Character or inst:IsDescendantOf(plr.Character)) then
            return true
        end
    end
    return false
end

local function cleanVisualEffectsLight()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if not isPlayerCharacter(obj) then
            if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke")
                or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("Beam") then
                pcall(function() obj.Enabled = false end)
            elseif obj:IsA("Explosion") then
                pcall(function() obj:Destroy() end)
            end
        end
    end
    for _, obj in ipairs(Lighting:GetChildren()) do
        -- Sky theme / Atmosphere / Clouds / AnomalySky ASLA bozma
        if obj.Name == "AnomalySky" or obj.Name == "AnomalyNukeCC"
            or obj:GetAttribute("AnomalySkyTheme")
            or obj:IsA("Sky") or obj:IsA("Atmosphere") or obj:IsA("Clouds") then
            -- koru
        elseif obj:IsA("BloomEffect") or obj:IsA("BlurEffect") or obj:IsA("SunRaysEffect")
            or obj:IsA("DepthOfFieldEffect") then
            pcall(function() obj.Enabled = false end)
        elseif obj:IsA("ColorCorrectionEffect") and not _G.AnomalySkyLocked then
            pcall(function() obj.Enabled = false end)
        end
    end
end

local function setFpsBoost(on)
    fpsBoostOn = on
    if fpsBoostConn then fpsBoostConn:Disconnect() fpsBoostConn = nil end

    if on then
        fpsBoostSaved = {
            globalShadows = Lighting.GlobalShadows,
            tech = nil,
        }
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        end)
        pcall(function()
            UserSettings():GetService("UserGameSettings").SavedQualityLevel = Enum.SavedQualitySetting.QualityLevel1
        end)
        pcall(function()
            settings().Rendering.MeshPartDetailLevel = Enum.MeshPartDetailLevel.Level01
        end)
        pcall(function()
            settings().Rendering.EagerBulkExecution = true
        end)
        pcall(function() Lighting.GlobalShadows = false end)
        pcall(function()
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
            Lighting.ShadowSoftness = 0
        end)
        pcall(function()
            Lighting.Technology = Enum.Technology.Legacy
        end)
        pcall(function()
            if workspace.Terrain then
                workspace.Terrain.WaterWaveSize = 0
                workspace.Terrain.WaterWaveSpeed = 0
                workspace.Terrain.WaterReflectance = 0
                workspace.Terrain.WaterTransparency = 1
                workspace.Terrain.Decoration = false
            end
        end)
        pcall(function()
            if setfpscap then setfpscap(999) end
        end)
        task.spawn(function()
            cleanVisualEffectsLight()
        end)

        local acc = 0
        fpsBoostConn = RunService.Heartbeat:Connect(function(dt)
            acc = acc + dt
            if acc < 3 then return end
            acc = 0
            pcall(function() Lighting.GlobalShadows = false end)
            pcall(function()
                if workspace.Terrain then
                    workspace.Terrain.WaterWaveSize = 0
                    workspace.Terrain.WaterWaveSpeed = 0
                    workspace.Terrain.Decoration = false
                end
            end)
            pcall(function()
                settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            end)
            -- yeni eklenen efektleri kapat
            for _, obj in ipairs(Lighting:GetChildren()) do
                if obj:IsA("BloomEffect") or obj:IsA("BlurEffect") or obj:IsA("SunRaysEffect")
                    or obj:IsA("DepthOfFieldEffect") then
                    if not _G.AnomalySkyLocked then
                        pcall(function() obj.Enabled = false end)
                    end
                end
            end
        end)
    else
        if fpsBoostSaved then
            pcall(function() Lighting.GlobalShadows = fpsBoostSaved.globalShadows end)
            fpsBoostSaved = nil
        end
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Automatic
        end)
    end
end

-- Nuke Ping Boost
local nukePingOn = false
local nukePingConn = nil
local nukeHidden = {}
local nukePingSaved = nil

local function setNukePingBoost(on)
    nukePingOn = on
    if nukePingConn then nukePingConn:Disconnect() nukePingConn = nil end

    if not on then
        for inst, _ in pairs(nukeHidden) do
            if inst and inst.Parent and inst:IsA("BasePart") then
                pcall(function() inst.LocalTransparencyModifier = 0 end)
            end
        end
        nukeHidden = {}
        pcall(function()
            local cc = Lighting:FindFirstChild("AnomalyNukeCC")
            if cc then cc:Destroy() end
        end)
        if nukePingSaved then
            pcall(function()
                Lighting.FogStart = nukePingSaved.fogStart
                Lighting.FogEnd = nukePingSaved.fogEnd
            end)
            nukePingSaved = nil
        end
        return
    end

    nukePingSaved = { fogStart = Lighting.FogStart, fogEnd = Lighting.FogEnd }

    pcall(function() workspace.StreamingEnabled = false end)
    pcall(function()
        if setsimulationradius then setsimulationradius(800) end
    end)
    pcall(function()
        sethiddenproperty(localPlayer, "MaxSimulationRadius", 800)
        sethiddenproperty(localPlayer, "SimulationRadius", 800)
    end)
    pcall(function()
        settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        settings().Rendering.MeshPartDetailLevel = Enum.MeshPartDetailLevel.Level01
        settings().Rendering.EagerBulkExecution = true
    end)
    pcall(function() Lighting.GlobalShadows = false end)

    -- sade renk
    pcall(function()
        if not _G.AnomalySkyLocked then
            local cc = Lighting:FindFirstChild("AnomalyNukeCC")
            if not cc then
                cc = Instance.new("ColorCorrectionEffect")
                cc.Name = "AnomalyNukeCC"
                cc.Parent = Lighting
            end
            cc.Saturation = -0.5
            cc.Contrast = 0.08
            cc.Brightness = 0.04
            cc.Enabled = true
        end
    end)

    -- 1 kez: uzaktaki part + efekt + animasyon temizliği
    task.spawn(function()
        local origin = Vector3.zero
        local hrp = localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
        if hrp then origin = hrp.Position end
        local spawnLoc = workspace:FindFirstChildOfClass("SpawnLocation")
        if spawnLoc then origin = spawnLoc.Position end
        for _, obj in ipairs(workspace:GetDescendants()) do
            if isPlayerCharacter(obj) then
                -- karakter animasyonlarına dokunma
            elseif obj:IsA("BasePart") then
                local d = (obj.Position - origin).Magnitude
                if d > 1000 then
                    pcall(function()
                        obj.LocalTransparencyModifier = 1
                        nukeHidden[obj] = true
                    end)
                end
            elseif obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke")
                or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("Beam") then
                pcall(function() obj.Enabled = false end)
            elseif obj:IsA("Sound") and not isPlayerCharacter(obj) then
                pcall(function() obj.Volume = 0 end)
            end
        end
    end)

    local acc = 0
    nukePingConn = RunService.Heartbeat:Connect(function(dt)
        acc = acc + dt
        if acc < 1.5 then return end
        acc = 0
        -- Sky theme sisini ASLA ezme
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
            Lighting.GlobalShadows = false
        end)
    end)
end

local function updateAllEsp()
    local myChar = localPlayer.Character
    local now = tick()

    for _, plr in ipairs(Players:GetPlayers()) do
        local char = plr.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local isLocal = (plr == localPlayer)

        if char and hum and hum.Health > 0 then
            local data = ensureEspForChar(char)
            local root = getCharRoot(char)

            -- Body / Box / Line sadece diğerleri
            if not isLocal then
                if data.highlight then
                    data.highlight.Enabled = bodyEspEnabled
                    data.highlight.FillColor = espBodyColor
                    data.highlight.OutlineColor = espBodyColor
                    data.highlight.OutlineTransparency = math.clamp(0.35 - (espBodyThickness - 1) * 0.1, 0.05, 0.5)
                    data.highlight.FillTransparency = math.clamp(0.65 - (espBodyThickness - 1) * 0.1, 0.35, 0.75)
                end

                if data.box then
                    data.box.Visible = boxEspEnabled
                    data.box.Color3 = espBoxColor
                    data.box.SurfaceColor3 = espBoxColor
                    data.box.LineThickness = 0.025 * espBoxScale
                end

                if data.beam then
                    data.beam.Enabled = lineEspEnabled
                    data.beam.Color = ColorSequence.new(espLineColor)
                    local w = 0.06 * espLineScale
                    data.beam.Width0 = w
                    data.beam.Width1 = w
                    local localRoot = myChar and getCharRoot(myChar)
                    if localRoot and data.lineAtt0 and data.lineAtt0.Parent ~= localRoot then
                        data.lineAtt0.Parent = localRoot
                        data.beam.Parent = localRoot
                    end
                    if root and data.lineAtt1 and data.lineAtt1.Parent ~= root then
                        data.lineAtt1.Parent = root
                    end
                elseif lineEspEnabled then
                    clearEspForChar(char)
                    ensureEspForChar(char)
                end

                -- Name ESP (ben hariç) + Roblox isim göstergesini kapat
                if data.nameGui and data.nameLbl then
                    data.nameGui.Enabled = nameEspEnabled
                    data.nameLbl.Text = plr.DisplayName ~= "" and plr.DisplayName or plr.Name
                    data.nameLbl.TextColor3 = Color3.fromRGB(240, 230, 255)
                end
                if hum then
                    if nameEspEnabled then
                        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                        hum.NameDisplayDistance = 0
                        pcall(function()
                            hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.DisplayWhenDamaged
                        end)
                    else
                        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
                        hum.NameDisplayDistance = 100
                    end
                end
            else
                if data.highlight then data.highlight.Enabled = false end
                if data.box then data.box.Visible = false end
                if data.beam then data.beam.Enabled = false end
                if data.nameGui then data.nameGui.Enabled = false end
            end

            -- Speed ESP (herkes + ben)
            if data.speedGui and data.speedLbl and root then
                data.speedGui.Enabled = speedEspEnabled
                if speedEspEnabled then
                    local pos = root.Position
                    local speed = 0
                    if data.lastPos and data.lastTime then
                        local dt = now - data.lastTime
                        if dt > 0 then
                            speed = (pos - data.lastPos).Magnitude / dt
                        end
                    end
                    data.lastPos = pos
                    data.lastTime = now
                    data.speedLbl.Text = string.format("%.0f", speed)
                    data.speedLbl.TextColor3 = Color3.fromRGB(200, 120, 255)
                end
            end
        else
            if char then clearEspForChar(char) end
        end
    end

    for char, _ in pairs(espObjects) do
        if not char.Parent then
            clearEspForChar(char)
        end
    end
end

task.spawn(function()
    while task.wait(0.15) do
        pcall(updateAllEsp)
    end
end)

local function makeEdgePart(parent, size, cf, color)
    local p = Instance.new("Part")
    p.Name = "EspBoxEdge"
    p.Anchored = true
    p.CanCollide = false
    p.Material = Enum.Material.Neon
    p.Color = color
    p.Size = size
    p.CFrame = cf
    p.Parent = parent
    return p
end

local function buildBoxEdges(parent, center, size, thickness, color)
    local parts = {}
    local hx, hy, hz = size.X / 2, size.Y / 2, size.Z / 2
    local t = thickness
    -- 12 edges of a rectangular box
    local edges = {
        -- bottom
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, -hy, -hz))},
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, -hy, hz))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(-hx, -hy, 0))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(hx, -hy, 0))},
        -- top
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, hy, -hz))},
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, hy, hz))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(-hx, hy, 0))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(hx, hy, 0))},
        -- vertical
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(-hx, 0, -hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(hx, 0, -hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(-hx, 0, hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(hx, 0, hz))},
    }
    for _, e in ipairs(edges) do
        table.insert(parts, makeEdgePart(parent, e[1], e[2], color))
    end
    return parts
end

local function refreshEspPreview()
    if espPreviewRefs.linePart then
        local s = 0.05 * espLineScale
        local z = espPreviewRefs.linePart.Size.Z
        espPreviewRefs.linePart.Size = Vector3.new(s, s, z)
        espPreviewRefs.linePart.Color = espLineColor
    end
    if espPreviewRefs.boxCenter and espPreviewRefs.boxSize then
        local world = EspViewport:FindFirstChildOfClass("WorldModel")
        if world then
            for _, p in ipairs(espPreviewRefs.boxParts) do
                if p and p.Parent then p:Destroy() end
            end
            local thickness = 0.035 * espBoxScale
            local pad = 0.15 * espBoxScale
            local sz = espPreviewRefs.boxSize + Vector3.new(pad, pad, pad)
            espPreviewRefs.boxParts = buildBoxEdges(world, espPreviewRefs.boxCenter, sz, thickness, espBoxColor)
        end
    end
    for _, p in ipairs(espPreviewRefs.bodyParts) do
        if p and p.Parent then
            local base = p:GetAttribute("BaseSize")
            if typeof(base) == "Vector3" then
                local expand = 0.12 * espBodyThickness
                p.Size = base + Vector3.new(expand, expand, expand)
            end
            p.Color = espBodyColor
            p.Transparency = math.clamp(0.4 - (espBodyThickness - 1) * 0.08, 0.2, 0.5)
        end
    end
end

local function buildEspPreview()
    local world = EspViewport:FindFirstChildOfClass("WorldModel")
    if world then world:Destroy() end
    world = Instance.new("WorldModel")
    world.Parent = EspViewport

    local cam = EspViewport:FindFirstChildOfClass("Camera")
    if not cam then
        cam = Instance.new("Camera")
        cam.Parent = EspViewport
    end
    EspViewport.CurrentCamera = cam

    -- Player: sol alt, sadece gövde/kollar/kafa görünsün, sağa baksın
    local playerModel
    local ok, result = pcall(function()
        return Players:CreateHumanoidModelFromUserId(localPlayer.UserId)
    end)
    if ok and result then
        playerModel = result
        for _, d in ipairs(playerModel:GetDescendants()) do
            if d:IsA("Script") or d:IsA("LocalScript") then d:Destroy() end
        end
        local hum = playerModel:FindFirstChildOfClass("Humanoid")
        if hum then hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end
    else
        playerModel = createBaconModel()
    end
    playerModel.Name = "EspPlayer"
    playerModel.Parent = world
    -- kameraya bakan, hafif sağa dönük (ters değil)
    local pPos = Vector3.new(-2.1, -0.7, 1.0)
    playerModel:PivotTo(CFrame.new(pPos) * CFrame.Angles(0, math.rad(-15), 0))

    -- Bacon: en sağ-üst, oyuncunun yüzüne baksın
    local bacon = createBaconModel()
    bacon.Parent = world
    local bPos = Vector3.new(2.55, 2.0, -0.5)
    bacon:PivotTo(CFrame.lookAt(bPos, pPos + Vector3.new(0, 1.2, 0)))

    local playerRoot = playerModel.PrimaryPart or playerModel:FindFirstChild("HumanoidRootPart") or playerModel:FindFirstChild("Torso") or playerModel:FindFirstChildWhichIsA("BasePart")
    local baconRoot = bacon.PrimaryPart or bacon:FindFirstChild("Torso") or bacon:FindFirstChildWhichIsA("BasePart")

    -- Line ESP
    local linePart = Instance.new("Part")
    linePart.Name = "EspLine"
    linePart.Anchored = true
    linePart.CanCollide = false
    linePart.Material = Enum.Material.Neon
    linePart.Color = espLineColor
    linePart.Parent = world
    if playerRoot and baconRoot then
        local a = playerRoot.Position + Vector3.new(0, 1.6, 0)
        local b = baconRoot.Position + Vector3.new(0, 1.3, 0)
        local mid = (a + b) / 2
        local dist = (a - b).Magnitude
        linePart.Size = Vector3.new(0.05 * espLineScale, 0.05 * espLineScale, dist)
        linePart.CFrame = CFrame.lookAt(mid, b)
    end

    -- Box ESP: içi boş dikdörtgen (12 kenar)
    local cf, size = bacon:GetBoundingBox()
    local boxCenter = cf.Position
    local boxSize = size
    local thickness = 0.035 * espBoxScale
    local pad = 0.15 * espBoxScale
    local boxParts = buildBoxEdges(world, boxCenter, boxSize + Vector3.new(pad, pad, pad), thickness, espBoxColor)

    -- Body ESP: karakter dışını saran belirgin neon/kabuk
    local bodyParts = {}
    for _, bp in ipairs(bacon:GetDescendants()) do
        if bp:IsA("BasePart") and bp.Name ~= "Hair" then
            local shell = Instance.new("Part")
            shell.Name = "BodyEspShell"
            shell.Anchored = true
            shell.CanCollide = false
            shell.Material = Enum.Material.Neon
            shell.Color = espBodyColor
            shell.Transparency = 0.35
            local expand = 0.12 * espBodyThickness
            shell.Size = bp.Size + Vector3.new(expand, expand, expand)
            shell.CFrame = bp.CFrame
            shell:SetAttribute("BaseSize", bp.Size)
            shell.Parent = world
            table.insert(bodyParts, shell)
        end
    end

    espPreviewRefs.linePart = linePart
    espPreviewRefs.boxParts = boxParts
    espPreviewRefs.bodyParts = bodyParts
    espPreviewRefs.baconModel = bacon
    espPreviewRefs.playerModel = playerModel
    espPreviewRefs.boxCenter = boxCenter
    espPreviewRefs.boxSize = boxSize

    -- Kamera: sol-alt oyuncu + sağ-üst bacon
    cam.CFrame = CFrame.new(Vector3.new(0.1, 1.6, 9.2), Vector3.new(0.15, 0.85, 0))
end

local function forceApplyLockedSky()
    if not lockedSkySettings or lockedSkySettings.name == "None" then return end
    if skyLockApplying then return end
    skyLockApplying = true
    local s = lockedSkySettings
    pcall(function()
        for _, ch in ipairs(Lighting:GetChildren()) do
            if ch:IsA("Sky") and ch.Name ~= "AnomalySky" then
                ch:Destroy()
            elseif ch:IsA("Atmosphere") or ch:IsA("Clouds") then
                ch:Destroy()
            end
        end
        local sky = Lighting:FindFirstChild("AnomalySky")
        if not sky then
            sky = Instance.new("Sky")
            sky.Name = "AnomalySky"
            sky.Parent = Lighting
        end
        sky.SkyboxBk = s.bk
        sky.SkyboxDn = s.dn
        sky.SkyboxFt = s.ft
        sky.SkyboxLf = s.lf
        sky.SkyboxRt = s.rt
        sky.SkyboxUp = s.up
        Lighting.ClockTime = s.clock
        Lighting.FogColor = s.fogColor
        -- Sky theme sis/fog her zaman uygulanır (boost/fullbright silmesin)
        Lighting.FogStart = s.fogStart
        Lighting.FogEnd = s.fogEnd
        if not (_G.AnomalyFullBright) then
            Lighting.OutdoorAmbient = s.outdoor
            pcall(function() Lighting.Ambient = s.outdoor end)
        else
            -- Full Bright: parlaklık artır, sky fog'u koru
            pcall(function()
                Lighting.Brightness = math.max(Lighting.Brightness, 2.5)
                Lighting.Ambient = Color3.fromRGB(180, 180, 180)
                Lighting.OutdoorAmbient = Color3.fromRGB(180, 180, 180)
            end)
        end
        pcall(function() Lighting.GeographicLatitude = 0 end)
    end)
    skyLockApplying = false
end

local function applySkyTheme(themeName)
    activeSkyTheme = themeName
    if themeName == "None" then
        lockedSkySettings = nil
        _G.AnomalySkyLocked = false
        if skyForceConn then skyForceConn:Disconnect(); skyForceConn = nil end
        if skyChildConn then skyChildConn:Disconnect(); skyChildConn = nil end
        pcall(function()
            local sky = Lighting:FindFirstChild("AnomalySky")
            if sky then sky:Destroy() end
            Lighting.ClockTime = 14
            Lighting.FogEnd = 100000
            Lighting.OutdoorAmbient = Color3.fromRGB(128, 128, 128)
        end)
        return
    end
    lockedSkySettings = buildSkySettings(themeName)
    _G.AnomalySkyLocked = true
    forceApplyLockedSky()
    if skyForceConn then skyForceConn:Disconnect() end
    skyForceConn = RunService.RenderStepped:Connect(function()
        if lockedSkySettings then
            forceApplyLockedSky()
        end
    end)
    if skyChildConn then skyChildConn:Disconnect() end
    skyChildConn = Lighting.ChildAdded:Connect(function(ch)
        if not lockedSkySettings then return end
        if ch:IsA("Sky") and ch.Name ~= "AnomalySky" then
            task.defer(function() pcall(function() ch:Destroy() end) end)
        elseif ch:IsA("Atmosphere") or ch:IsA("Clouds") then
            task.defer(function() pcall(function() ch:Destroy() end) end)
        end
    end)
    -- oyun lighting değiştirirse geri al (anan.txt)
    pcall(function()
        for _, prop in ipairs({"ClockTime", "FogColor", "FogEnd", "FogStart", "OutdoorAmbient", "Ambient", "Brightness"}) do
            Lighting:GetPropertyChangedSignal(prop):Connect(function()
                if lockedSkySettings and not skyLockApplying then
                    forceApplyLockedSky()
                end
            end)
        end
    end)
end

local function loadAssetImage(targetImageLabel, id)
    targetImageLabel.Image = "rbxassetid://" .. tostring(id)
    local InsertService = game:GetService("InsertService")
    
    pcall(function()
        local result = InsertService:LoadAsset(id)
        if result then
            local decal = result:FindFirstChildOfClass("Decal") or result:FindFirstChildWhichIsA("Decal", true)
            if decal and decal.Texture then
                targetImageLabel.Image = decal.Texture
                return
            end
        end
    end)

    pcall(function()
        local rawData = game:HttpGet("https://assetdelivery.roblox.com/v1/asset/?id=" .. tostring(id))
        if rawData then
            local realId = rawData:match('<url>http[s]?://.-id=(%d+)</url>') or rawData:match('id=(%d+)')
            if realId then
                targetImageLabel.Image = "rbxassetid://" .. realId
            end
        end
    end)
end

task.spawn(function()
    loadAssetImage(BgImageLabel, bgAssetId)
    loadAssetImage(OldImageLabel, oldAssetId)
    loadAssetImage(ToggleButton, oldAssetId)
end)
