    end

    if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
        drag_view_dragInput = input
    end
end)

        local delta = input.Position - drag_view_dragStart

            drag_view_startPos.X.Scale,
            drag_view_startPos.X.Offset + delta.X,
            drag_view_startPos.Y.Scale,
            drag_view_startPos.Y.Offset + delta.Y
        )
    end
end)

UserInputService.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        local mousePos = input.Position
        local pos = OuterFrame_View.AbsolutePosition
        local size = OuterFrame_View.AbsoluteSize
        
        if mousePos.X >= pos.X and mousePos.X <= pos.X + size.X and
           mousePos.Y >= pos.Y and mousePos.Y <= pos.Y + size.Y then
            if OuterFrame_View.Visible then
                resetInactivity_View()
            end
        end

local function loadAssetImage(targetImageLabel, id)
    targetImageLabel.Image = "rbxassetid://" .. tostring(id)
    local InsertService = game:GetService("InsertService")
    
    pcall(function()
        local result = InsertService:LoadAsset(id)
        if result then
            local decal = result:FindFirstChildOfClass("Decal") or result:FindFirstChildWhichIsA("Decal", true)
            if decal and decal.Texture then
                targetImageLabel.Image = decal.Texture
                return
            end
        end
    end)

    pcall(function()
        local rawData = game:HttpGet("https://assetdelivery.roblox.com/v1/asset/?id=" .. tostring(id))
        if rawData then
            local realId = rawData:match('<url>http[s]?://.-id=(%d+)</url>') or rawData:match('id=(%d+)')
            if realId then
                targetImageLabel.Image = "rbxassetid://" .. realId
            end
        end
    end)
end

task.spawn(function()
    loadAssetImage(BgImageLabel, bgAssetId)
    loadAssetImage(OldImageLabel, oldAssetId)
    loadAssetImage(ToggleButton, oldAssetId)
end)
