local function makeDraggable(gui)
    local dragging, dragInput, dragStart, startPos
    gui.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = gui.Position
            
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)
    gui.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            gui.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
end

local antiAntiDesyncConnections = {}
local antiAntiDesyncActive = false
local dropBRActive = false

local function zeroHrpVelocity()
    local char = LocalPlayer.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    if root then
        ignoreVelocitySpoof = true
        root.AssemblyLinearVelocity = Vector3.zero
        root.AssemblyAngularVelocity = Vector3.zero
        ignoreVelocitySpoof = false
    end
end

local function stopAntiAntiDesyncInternal()
    antiAntiDesyncActive = false
    -- Önce hızı 0 yap, sonra bağlantıları kes
    zeroHrpVelocity()
    for _, conn in ipairs(antiAntiDesyncConnections) do
        if conn then
            if typeof(conn) == "RBXScriptConnection" then
                pcall(function() conn:Disconnect() end)
            elseif typeof(conn) == "thread" then
                pcall(function() task.cancel(conn) end)
            end
        end
    end
    table.clear(antiAntiDesyncConnections)
    zeroHrpVelocity()
end

local function startAntiAntiDesyncInternal(forceValue)
    stopAntiAntiDesyncInternal()
    antiAntiDesyncActive = true
    -- Klasik walkfling tarzı: yüksek çarpan + X/Z güç (insanları uçurur)
    local force = forceValue or (-9e25 * 10)

    local LP = LocalPlayer

    -- Diğer oyuncuların CanCollide'ını kapat (içinden geçip fling için)
    local collConn = RunService.Stepped:Connect(function()
        if not antiAntiDesyncActive then return end
        for _, plr in ipairs(Players:GetPlayers()) do
            if plr ~= LP and plr.Character then
                for _, part in ipairs(plr.Character:GetDescendants()) do
                    if part:IsA("BasePart") then
                        part.CanCollide = false
                    end
                end
            end
        end
    end)
    table.insert(antiAntiDesyncConnections, collConn)

    -- Walkfling: yüksek velocity pulse → restore → küçük salınım (diğerlerini uçurur)
    local flingThread = task.spawn(function()
        local movel = 0.1
        while antiAntiDesyncActive do
            RunService.Heartbeat:Wait()
            if not antiAntiDesyncActive then break end

            local character = LP.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if not (character and character.Parent and root and root.Parent) then
                -- karakter yoksa bekle
            else
                local vel = root.AssemblyLinearVelocity
                -- X/Z yüksek güç + mevcut velocity * 10000
                ignoreVelocitySpoof = true
                root.AssemblyLinearVelocity = Vector3.new(
                    vel.X * 10000 + force,
                    vel.Y,
                    vel.Z * 10000 + force
                )
                ignoreVelocitySpoof = false

                RunService.RenderStepped:Wait()
                if character and character.Parent and root and root.Parent then
                    ignoreVelocitySpoof = true
                    root.AssemblyLinearVelocity = vel
                    ignoreVelocitySpoof = false
                end

                RunService.Stepped:Wait()
                if character and character.Parent and root and root.Parent then
                    ignoreVelocitySpoof = true
                    root.AssemblyLinearVelocity = vel + Vector3.new(movel, 0, movel)
                    ignoreVelocitySpoof = false
                    movel = movel * -1
                end
            end
        end
    end)
    table.insert(antiAntiDesyncConnections, flingThread)
end

local function setAntiAntiDesync(state)
    antiAntiDesyncEnabled = state
    AntiAntiDesyncBtn.Text = state and "ON" or "OFF"
    AntiAntiDesyncBtn.BackgroundColor3 = state and Color3.fromRGB(130, 50, 210) or Color3.fromRGB(55, 45, 75)
    AntiAntiDesyncBtn.TextColor3 = state and Color3.fromRGB(255, 255, 255) or Color3.fromRGB(180, 180, 180)

    if state then
        startAntiAntiDesyncInternal(-9e25 * 10)
    else
        stopAntiAntiDesyncInternal()
    end
end

    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Parent = ScreenGui
    
    local row = math.floor(layoutCounter / 2)
    local col = layoutCounter % 2

    btn.TextSize = 10

    btn.ZIndex = 15

    local stroke = Instance.new("UIStroke", btn)
    stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    stroke.Thickness = 1.5
    
    updateFloatVisuals(btn, false)
    floatButtons[name] = btn
    makeDraggable(btn)
    
    if isToggle then
        floatToggles[name] = false
        btn.MouseButton1Click:Connect(function() setFloatToggle(name, not floatToggles[name]) end)
    else
        btn.MouseButton1Click:Connect(function()
            updateFloatVisuals(btn, true)
            task.spawn(function() task.wait(0.3); updateFloatVisuals(btn, false) end)
            
            local char = LocalPlayer.Character
            local hrp = char and char:FindFirstChild("HumanoidRootPart")
            local humanoid = char and char:FindFirstChildOfClass("Humanoid")
            
            if name == "Reset" and hrp and humanoid then
                -- Anti Anti Desync açıksa geçici durdur (Reset çalışsın)
                local wasAntiOn = antiAntiDesyncEnabled and antiAntiDesyncActive
                if wasAntiOn then
                    stopAntiAntiDesyncInternal()
                end

                ignoreVelocitySpoof = true
                isResetting = true
                
                if resetMode == "Tp High" then
                    Camera.CameraType = Enum.CameraType.Scriptable
                    local savedCFrame = Camera.CFrame
                    Camera.CFrame = savedCFrame

                    task.spawn(function()
                        while ignoreVelocitySpoof do
                            if hrp and hrp.Parent then
                                ignoreVelocitySpoof = true
                                hrp.AssemblyLinearVelocity = Vector3.new(0, 3e6, 0)
                            end
                            task.wait(0.3)
                        end
                    end)

                    local respawnConn
                    respawnConn = LocalPlayer.CharacterAdded:Connect(function(newChar)
                        ignoreVelocitySpoof = false
                        isResetting = false
                        task.wait(0.1)
                        Camera.CameraType = Enum.CameraType.Custom
                        if newChar:FindFirstChildOfClass("Humanoid") then
                            Camera.CameraSubject = newChar:FindFirstChildOfClass("Humanoid")
                        end
                        -- Reset bittikten sonra Anti Anti Desync'i geri aç
                        if wasAntiOn and antiAntiDesyncEnabled then
                            startAntiAntiDesyncInternal(-9e25 * 10)
                        end
                        respawnConn:Disconnect()
                    end)
                elseif resetMode == "Fly" then
                    Camera.CameraType = Enum.CameraType.Custom
                    if humanoid then Camera.CameraSubject = humanoid end
                    
                    task.spawn(function()
                        while ignoreVelocitySpoof do
                            if hrp and hrp.Parent then
                                ignoreVelocitySpoof = true
                                hrp.AssemblyLinearVelocity = Vector3.new(0, 3e6, 0)
                            end
                            task.wait(0.3)
                        end
                    end)
                    task.delay(1.5, function() 
                        ignoreVelocitySpoof = false 
                        isResetting = false
                        if wasAntiOn and antiAntiDesyncEnabled then
                            startAntiAntiDesyncInternal(-9e25 * 10)
                        end
                    end)
                end
            end
            
            if name == "Tp Down" and hrp then
                triggerTpDown(hrp)
            end

            -- DROP BRAINROT: X/Z 100k fling (SENİ UÇURMAZ), 0.2s total, 0.1s'de güç 0
            if name == "Drop Brainrot" then
                local wasAntiOn = antiAntiDesyncEnabled and antiAntiDesyncActive
                if wasAntiOn then
                    stopAntiAntiDesyncInternal()
                end
                dropBRActive = true

                local dropChar = LocalPlayer.Character
                local dropHrp = dropChar and dropChar:FindFirstChild("HumanoidRootPart")
                if dropHrp then
                    local savedCF = dropHrp.CFrame
                    local forceXZ = 100000
                    local startT = tick()
                    local dropConn, collConn, lockConn

                    -- Diğer oyuncular CanCollide off (onlar uçsun)
                    collConn = RunService.Stepped:Connect(function()
                        if not dropBRActive then return end
                        for _, plr in ipairs(Players:GetPlayers()) do
                            if plr ~= LocalPlayer and plr.Character then
                                for _, part in ipairs(plr.Character:GetDescendants()) do
                                    if part:IsA("BasePart") then
                                        part.CanCollide = false
                                    end
                                end
                            end
                        end
                    end)

                    -- Her frame pozisyonu sabitle → seni uçurmaz
                    lockConn = RunService.RenderStepped:Connect(function()
                        if not dropBRActive then return end
                        local h = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                        if h then
                            ignoreVelocitySpoof = true
                            h.CFrame = savedCF
                            h.AssemblyAngularVelocity = Vector3.zero
                            ignoreVelocitySpoof = false
                        end
                    end)

                    dropConn = RunService.Heartbeat:Connect(function()
                        if not dropBRActive then
                            if dropConn then dropConn:Disconnect() end
                            if collConn then collConn:Disconnect() end
                            if lockConn then lockConn:Disconnect() end
                            return
                        end
                        local elapsed = tick() - startT
                        local char = LocalPlayer.Character
                        local hrp = char and char:FindFirstChild("HumanoidRootPart")
                        if not hrp then
                            dropBRActive = false
                            if dropConn then dropConn:Disconnect() end
                            if collConn then collConn:Disconnect() end
                            if lockConn then lockConn:Disconnect() end
                            return
                        end

                        if elapsed >= 0.2 then
                            dropBRActive = false
                            if dropConn then dropConn:Disconnect() end
                            if collConn then collConn:Disconnect() end
                            if lockConn then lockConn:Disconnect() end
                            ignoreVelocitySpoof = true
                            hrp.AssemblyLinearVelocity = Vector3.zero
                            hrp.AssemblyAngularVelocity = Vector3.zero
                            hrp.CFrame = savedCF
                            ignoreVelocitySpoof = false
                            if wasAntiOn and antiAntiDesyncEnabled then
                                startAntiAntiDesyncInternal(-9e25 * 10)
                            end
                            return
                        end

                        ignoreVelocitySpoof = true
                        if elapsed < 0.1 then
                            -- X/Z 100k — collision impulse için, pozisyon lockConn ile sabit
                            hrp.AssemblyLinearVelocity = Vector3.new(forceXZ, 0, forceXZ)
                        else
                            -- 0.1s sonrası güç 0
                            hrp.AssemblyLinearVelocity = Vector3.zero
                            hrp.AssemblyAngularVelocity = Vector3.zero
                        end
                        ignoreVelocitySpoof = false
                    end)
                else
                    dropBRActive = false
                    if wasAntiOn and antiAntiDesyncEnabled then
                        startAntiAntiDesyncInternal(-9e25 * 10)
                    end
                end
            end
        end)
    end
    layoutCounter = layoutCounter + 1
end

local function doBatCounterOn(attacker)
    if not batCounterEnabled or batCounterBusy then return end
    if tick() - lastBatCounterFire < 0.5 then return end
    lastBatCounterFire = tick()
    batCounterBusy = true
    local char = LocalPlayer.Character
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    local hrp = char and char:FindFirstChild("HumanoidRootPart")
    local targetHRP = attacker.Character and attacker.Character:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp or not targetHRP then batCounterBusy = false return end
    local lookStart = tick()
    local lookConn
    lookConn = RunService.RenderStepped:Connect(function()
        if not hrp.Parent or not targetHRP.Parent or tick() - lookStart > 0.1 then

        local bat = findBatTool()
        if bat and hum then
            pcall(function()
                if bat.Parent ~= char then hum:EquipTool(bat) end
                bat:Activate()
            end)
        end
    end)
    task.delay(0.12, function()
        if lookConn then lookConn:Disconnect() end
        batCounterBusy = false
    end)
end

        if PM then

        end
    end)
end

local lastMoveDir = Vector3.zero
local wasStealing = false
local lastBatAttack = 0

RunService.RenderStepped:Connect(function()
    local char = LocalPlayer.Character
    if not char then return end
    local hum = char:FindFirstChildOfClass("Humanoid")
    local hrp = char:FindFirstChild("HumanoidRootPart")
    if not hum or not hrp then return end

    if safeModeEnabled and hum.WalkSpeed <= 0 then
        lastMoveDir = Vector3.zero; clearLinVel(); return 
    end

    local st = hum:GetState()
    if hum.PlatformStand or st == Enum.HumanoidStateType.Physics or st == Enum.HumanoidStateType.Ragdoll then
        lastMoveDir = Vector3.zero; clearLinVel(); return
    end

    ensureLinVel()

    if autoTpDownEnabled and not isResetting then
        local thresholdLimit = spawnHeight + (tonumber(AutoTpDownInput.Text) or 20)
        if hrp.Position.Y > thresholdLimit then
            if not wasAboveLimit then
                wasAboveLimit = true
                triggerTpDown(hrp)
            end
        else
            wasAboveLimit = false
        end
    end

    -- Bat Aimbot
    if floatToggles["Bat Aimbot"] then
        local target = getNearestPlayer()
        if target and target.Character and target.Character:FindFirstChild("HumanoidRootPart") then
            local tHrp = target.Character.HumanoidRootPart
            local dir = tHrp.Position - hrp.Position
            local dir2D = Vector3.new(dir.X, 0, dir.Z)
            if dir2D.Magnitude > 0 then
                local u = dir2D.Unit
                setHrpVelocity(hrp, Vector3.new(u.X * 65, hrp.AssemblyLinearVelocity.Y, u.Z * 65))
                setLinVelXZ(u.X * 65, u.Z * 65)
            end
            
            local now = tick()
            if now - lastBatAttack >= 0.7 then
                lastBatAttack = now
                local bat = char:FindFirstChild("Bat")
                if not bat then
                    local bp = LocalPlayer.Backpack:FindFirstChild("Bat")
                    if bp and hum then
                        hum:EquipTool(bp)
                        bat = bp
                    end
                end
                if bat then
                    bat:Activate()
                end
            end

            -- Mirror Tp Down logic
            if mirrorTpDownEnabled then
                local currentY = tHrp.Position.Y
                if lastTargetY ~= nil then
                    local deltaY = lastTargetY - currentY
                    if deltaY > 4 and not mirrorTpDownFired then
                        mirrorTpDownFired = true
                        triggerTpDown(hrp)
                        task.delay(1.5, function() mirrorTpDownFired = false end)
                    end
                end
                lastTargetY = currentY
            else
                lastTargetY = nil
                mirrorTpDownFired = false
            end

            return 
        end
    else
        lastTargetY = nil
        mirrorTpDownFired = false
    end

    local isCurrentlyStealing = false
    if LocalPlayer:GetAttribute("Stealing") == true or LocalPlayer:GetAttribute("Steal") == true then isCurrentlyStealing = true end
    if not isCurrentlyStealing and (char:GetAttribute("Stealing") == true or char:GetAttribute("Steal") == true) then isCurrentlyStealing = true end

    if isCurrentlyStealing and not wasStealing then
        -- Stealing moduna geçti → countdown göster
        showStealCountdown()
        if autoSwitchEnabled then
            if floatToggles["Lagger Normal"] then 
                setFloatToggle("Lagger Steal", true)
            else 
                setFloatToggle("Steal Speed", true) 
            end
        end
    end

-- FLOAT (Ambitious-style: rise + hold Y, cycle)
----------------------------------------------------
local FLOAT = {
    RISE = 17,
    RATE = 55,
    CYCLE = 2.5,
    PAUSE = 0.5,
    conn = nil,
    thread = nil,
    running = false,
    locked = false,
    targetY = nil,
}

local function floatGetHRP()
    local c = LocalPlayer.Character
    local hrp = c and c:FindFirstChild("HumanoidRootPart")
    local hum = c and c:FindFirstChildOfClass("Humanoid")
    return hrp, hum
end

local function floatStopLoop(doTpDown)
    if FLOAT.conn then
        pcall(function() FLOAT.conn:Disconnect() end)
        FLOAT.conn = nil
    end
    FLOAT.locked = false
    FLOAT.targetY = nil
    if doTpDown then
        local hrp = floatGetHRP()
        if hrp then triggerTpDown(hrp) end
    end
end

local function floatStartLoop()
    if FLOAT.conn then return end
    FLOAT.locked = false
    FLOAT.targetY = nil
    local _, hum0 = floatGetHRP()
    if hum0 then pcall(function() hum0:ChangeState(Enum.HumanoidStateType.Freefall) end) end

    FLOAT.conn = RunService.Heartbeat:Connect(function(dt)
        local root = floatGetHRP()
        if not root then return end
        local pos = root.Position
        local vel = root.AssemblyLinearVelocity
        local rot = root.CFrame - root.CFrame.Position

        if not FLOAT.targetY then
            FLOAT.targetY = pos.Y + FLOAT.RISE
        end

        if not FLOAT.locked then
            local newY = pos.Y + FLOAT.RATE * dt
            if newY >= FLOAT.targetY then
                newY = FLOAT.targetY
                FLOAT.locked = true
            end
            ignoreVelocitySpoof = true
            root.CFrame = CFrame.new(pos.X, newY, pos.Z) * rot
            root.AssemblyLinearVelocity = Vector3.new(vel.X, 0, vel.Z)
            ignoreVelocitySpoof = false
        else
            -- Hold Y
            ignoreVelocitySpoof = true
            root.CFrame = CFrame.new(pos.X, FLOAT.targetY, pos.Z) * rot
            root.AssemblyLinearVelocity = Vector3.new(vel.X, 0, vel.Z)
            ignoreVelocitySpoof = false
        end
    end)
end

local function floatWaitGrounded(timeout)
    local deadline = os.clock() + (timeout or 3)
    while os.clock() < deadline do
        if not FLOAT.running then return end
        local hrp, hum = floatGetHRP()
        if not hrp or not hum or hum.Health <= 0 then return end
        if hum.FloorMaterial ~= Enum.Material.Air then
            task.wait(FLOAT.PAUSE)
            return
        end
        RunService.Heartbeat:Wait()
    end
end

local function floatStop()
    FLOAT.running = false
    if FLOAT.thread then pcall(task.cancel, FLOAT.thread); FLOAT.thread = nil end
    floatStopLoop(true)
    floatEnabled = false
    FloatBtn.Text = "OFF"
    FloatBtn.BackgroundColor3 = Color3.fromRGB(55, 45, 75)
    FloatBtn.TextColor3 = Color3.fromRGB(180, 180, 180)
    if floatToggles["Float"] ~= nil then
        floatToggles["Float"] = false
        if floatButtons["Float"] then updateFloatVisuals(floatButtons["Float"], false) end
    end
end

local function floatStart()
    if FLOAT.running then return end
    FLOAT.running = true
    floatEnabled = true
    FloatBtn.Text = "ON"
    FloatBtn.BackgroundColor3 = Color3.fromRGB(130, 50, 210)
    FloatBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    if floatToggles["Float"] ~= nil then
        floatToggles["Float"] = true
        if floatButtons["Float"] then updateFloatVisuals(floatButtons["Float"], true) end
    end
    floatStartLoop()
    FLOAT.thread = task.spawn(function()
        while FLOAT.running do
            local t0 = os.clock()
            while FLOAT.running and (os.clock() - t0) < FLOAT.CYCLE do
                task.wait(0.1)
            end
            if not FLOAT.running then break end
            floatStopLoop(true)
            floatWaitGrounded(3)
            if not FLOAT.running then break end
            floatStartLoop()
        end
    end)
end

FloatBtn.MouseButton1Click:Connect(function()
    if FLOAT.running then floatStop() else floatStart() end
end)

if floatButtons["Float"] then
    -- Override the generic toggle to use floatStart/Stop
    floatButtons["Float"].MouseButton1Click:Connect(function()
        task.defer(function()
            if floatToggles["Float"] then
                if not FLOAT.running then floatStart() end
            else
                if FLOAT.running then floatStop() end
            end
        end)
    end)
end

local currentBatSkin = "DiamondSword"
local currentMedusaSkin = "GoldenDesertEagle"
local batSkinKeys = {"DiamondSword", "Katana"}
local medusaSkinKeys = {"GoldenDesertEagle", "Skull"}
local batSkinIdx, medusaSkinIdx = 1, 1

local function applyToolSkin(tool, skinKey)
    local def = SKIN_ASSETS[skinKey]
    if not def or not tool then return end
    local handle = tool:FindFirstChild("Handle")
    if not handle then return end
    pcall(function()
        for _, d in ipairs(handle:GetChildren()) do
            if d:IsA("SpecialMesh") or d:IsA("MeshPart") then d:Destroy() end
        end
        local mesh = Instance.new("SpecialMesh")
        mesh.MeshId = def.mesh
        mesh.TextureId = def.tex
        mesh.Scale = def.scale
        mesh.Parent = handle
        local motor = handle:FindFirstChildOfClass("Motor6D")
        if motor then
            motor.C0 = def.c0
        end
        -- Hide original mesh parts
        for _, d in ipairs(tool:GetDescendants()) do
            if d:IsA("BasePart") and d ~= handle then
                d.Transparency = 1
                d.LocalTransparencyModifier = 1
            end
            if d:IsA("Decal") or d:IsA("Texture") then
                d.Transparency = 1
            end
        end
    end)
end

local function reapplySkins()
    local function scan(container)
        if not container then return end
        for _, t in ipairs(container:GetChildren()) do
            if t:IsA("Tool") then
                local n = t.Name:lower()
                if n:find("bat") then applyToolSkin(t, currentBatSkin) end
                if n:find("medusa") then applyToolSkin(t, currentMedusaSkin) end
            end
        end
    end
    scan(LocalPlayer:FindFirstChild("Backpack"))
    scan(LocalPlayer.Character)
end
