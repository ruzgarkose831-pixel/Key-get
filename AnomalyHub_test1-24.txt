-- ========== SAĞ TARAF KALICI / SÜRÜKLENEBİLİR KARE BUTONLAR ==========
do
    local sideStates = {
        AutoLeft = false,
        AutoRight = false,
        BatAimbot = false,
        StealSpeed = false,
        TpBat = false,
        LaggerNormal = false,
        LaggerSteal = false,
        AutoPlay = false,
    }

    local sideButtons = {} -- name -> TextButton
    local offColor = Color3.fromRGB(22, 22, 30)
    local onColor = Color3.fromRGB(160, 30, 255)
    local offText = Color3.fromRGB(190, 190, 210)
    local onText = Color3.fromRGB(255, 255, 255)

    local BTN = 54 -- tüm butonlar eşit kare
    local GAP = 5
    -- varsayılan grid sağ üst, Y offset -1
    local baseTop = 0 -- 1 pixel aşağı (-1 + 1)

    local function styleBtn(btn, active)
        if not btn or not btn.Parent then return end
        if active then
            TweenService:Create(btn, TweenInfo.new(0.15), {
                BackgroundColor3 = onColor,
                BackgroundTransparency = 0.35,
                TextColor3 = onText,
            }):Play()
            local s = btn:FindFirstChildOfClass("UIStroke")
            if s then s.Color = Color3.fromRGB(220, 130, 255) s.Transparency = 0.05 end
        else
            TweenService:Create(btn, TweenInfo.new(0.15), {
                BackgroundColor3 = Color3.fromRGB(28, 28, 40),
                BackgroundTransparency = 0.55,
                TextColor3 = Color3.fromRGB(230, 230, 255),
            }):Play()
            local s = btn:FindFirstChildOfClass("UIStroke")
            if s then s.Color = Color3.fromRGB(170, 70, 255) s.Transparency = 0.2 end
        end
    end

    local function setOff(name)
        sideStates[name] = false
        if sideButtons[name] then styleBtn(sideButtons[name], false) end
    end

    local function makeDraggable(btn)
        local dragging = false
        local dragInput, dragStart, startPos
        local moved = false

        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                moved = false
                dragStart = input.Position
                startPos = btn.Position
                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end)
            end
        end)

        btn.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
        end)

        UserInputService.InputChanged:Connect(function(input)
            if input == dragInput and dragging then
                local delta = input.Position - dragStart
                if math.abs(delta.X) > 4 or math.abs(delta.Y) > 4 then
                    moved = true
                end
                btn.Position = UDim2.new(
                    startPos.X.Scale,
                    startPos.X.Offset + delta.X,
                    startPos.Y.Scale,
                    startPos.Y.Offset + delta.Y
                )
            end
        end)

        return function()
            return moved
        end
    end

    local function makeSideBtn(name, text, col, row, flashOnly)
        -- sağa yaslı scale pozisyon (her cihazda görünür)
        local xOff = -((2 - col) * (BTN + GAP) + 8)
        local yOff = baseTop + row * (BTN + GAP)

        local btn = Instance.new("TextButton")
        btn.Name = "AnomalySide_" .. name
        btn.Size = UDim2.new(0, BTN, 0, BTN)
        btn.Position = UDim2.new(1, xOff, 0, yOff)
        btn.AnchorPoint = Vector2.new(0, 0)
        btn.BackgroundColor3 = Color3.fromRGB(28, 28, 40)
        btn.BackgroundTransparency = 0.55
        btn.BorderSizePixel = 0
        btn.Text = text
        btn.TextColor3 = Color3.fromRGB(230, 230, 255)
        btn.Font = Enum.Font.GothamBold
        btn.TextSize = (string.len(text) > 10) and 9 or 11
        btn.TextWrapped = true
        btn.TextStrokeTransparency = 1
        btn.TextStrokeColor3 = Color3.fromRGB(0, 0, 0)
        btn.AutoButtonColor = false
        btn.Visible = true
        btn.ZIndex = 50 -- intro (100+) arkasında
        btn.Active = true
        btn.Parent = ScreenGui

        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, 10)
        corner.Parent = btn

        -- yazı çevresi değil, butonun dışı çerçeve
        local stroke = Instance.new("UIStroke")
        stroke.Color = Color3.fromRGB(170, 70, 255)
        stroke.Thickness = 1.6
        stroke.Transparency = 0.2
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        stroke.Parent = btn

        sideButtons[name] = btn
        local wasMoved = makeDraggable(btn)

        if flashOnly then
            btn.MouseButton1Click:Connect(function()
                if wasMoved() then return end
                styleBtn(btn, true)
                task.delay(0.3, function()
                    if btn and btn.Parent then styleBtn(btn, false) end
                end)
            end)
        else
            btn.MouseButton1Click:Connect(function()
                if wasMoved() then return end
                local newState = not sideStates[name]

                if name == "AutoLeft" and newState then
                    setOff("AutoRight")
                    setOff("BatAimbot")
                elseif name == "AutoRight" and newState then
                    setOff("AutoLeft")
                    setOff("BatAimbot")
                elseif name == "BatAimbot" and newState then
                    setOff("AutoLeft")
                    setOff("AutoRight")
                    setOff("AutoPlay")
                elseif name == "AutoPlay" and newState then
                    setOff("BatAimbot")
                elseif name == "StealSpeed" and newState then
                    setOff("LaggerNormal")
                    setOff("LaggerSteal")
                elseif (name == "LaggerNormal" or name == "LaggerSteal") and newState then
                    setOff("StealSpeed")
                end

                sideStates[name] = newState
                styleBtn(btn, newState)
            end)
        end

        return btn
    end

    local function destroySideButtons()
        for name, btn in pairs(sideButtons) do
            if btn then btn:Destroy() end
        end
        sideButtons = {}
        sideStates.AutoLeft = false
        sideStates.AutoRight = false
        sideStates.AutoPlay = false
        sideStates.BatAimbot = false
        sideStates.StealSpeed = false
        sideStates.TpBat = false
        sideStates.LaggerNormal = false
        sideStates.LaggerSteal = false
    end

    local function rebuildSidePanel(autoPlayMode)
        destroySideButtons()

        local row = 0
        if autoPlayMode then
            -- en sol üst: tek Auto Play
            makeSideBtn("AutoPlay", "Auto Play", 0, row, false)
            row = row + 1
        else
            makeSideBtn("AutoLeft", "Auto Left", 0, row, false)
            makeSideBtn("AutoRight", "Auto Right", 1, row, false)

local function createWhiteDummy()
    local model = Instance.new("Model")
    model.Name = "Dummy"

    local function makePart(name, size, cf, color)
        local p = Instance.new("Part")
        p.Name = name
        p.Size = size

        p.Anchored = true
        p.CanCollide = false
        p.Material = Enum.Material.SmoothPlastic

        p.Parent = model
        return p
    end

    local torso = makePart("Torso", Vector3.new(2, 2, 1), CFrame.new(0, 1, 0))

    face.Face = Enum.NormalId.Front

    face.Parent = head

    -- slight left turn already applied via camera / model CFrame later
    model.PrimaryPart = torso
    return model
end

local function setupViewport(viewport, usePlayerAvatar)
    local world = Instance.new("WorldModel")
    world.Parent = viewport

    local cam = Instance.new("Camera")
    cam.Parent = viewport
    viewport.CurrentCamera = cam

    local model
    if usePlayerAvatar then
        local ok, result = pcall(function()
            return Players:CreateHumanoidModelFromUserId(localPlayer.UserId)
        end)
        if ok and result then
            model = result
            model.Parent = world
            -- remove scripts
            for _, d in ipairs(model:GetDescendants()) do
                if d:IsA("Script") or d:IsA("LocalScript") then
                    d:Destroy()
                end
            end
            local hum = model:FindFirstChildOfClass("Humanoid")
            if hum then hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end
        end
    end

    if not model then
        model = createWhiteDummy()
        model.Parent = world
    end

    -- position & slight left yaw
    local primary = model.PrimaryPart or model:FindFirstChild("HumanoidRootPart") or model:FindFirstChild("Torso") or model:FindFirstChildWhichIsA("BasePart")
    if primary then
        -- yüz kameraya baksın, hafif sola dönük
        model:PivotTo(CFrame.new(0, 0, 0) * CFrame.Angles(0, math.rad(155), 0))
        local bbox = model:GetExtentsSize()
        local dist = math.max(bbox.X, bbox.Y, bbox.Z) * 1.35
        cam.CFrame = CFrame.new(Vector3.new(-0.35, bbox.Y * 0.35, dist), Vector3.new(0, bbox.Y * 0.28, 0))
    else
        cam.CFrame = CFrame.new(0, 1.5, 5)
    end
end

-- FPS Boost: ownership + partikül/gölge/efekt/su temizliği
local fpsBoostOn = false
local fpsBoostConn = nil
local fpsBoostSaved = nil

local function cleanVisualEffects(root)
    for _, obj in ipairs(root:GetDescendants()) do
        if obj:IsA("ParticleEmitter") or obj:IsA("Trail") or obj:IsA("Smoke")
            or obj:IsA("Fire") or obj:IsA("Sparkles") or obj:IsA("Beam") then
            pcall(function() obj.Enabled = false end)
            pcall(function() obj:Destroy() end)
        elseif obj:IsA("Explosion") then
            pcall(function() obj:Destroy() end)
        elseif obj:IsA("PostEffect") or obj:IsA("BloomEffect") or obj:IsA("BlurEffect")
            or obj:IsA("SunRaysEffect") or obj:IsA("ColorCorrectionEffect")
            or obj:IsA("DepthOfFieldEffect") then
            -- sky theme efektlerini silme
            if not obj:GetAttribute("AnomalySkyTheme") then
                pcall(function() obj.Enabled = false end)
            end
        end
    end
end

local function setFpsBoost(on)
    fpsBoostOn = on
    if fpsBoostConn then fpsBoostConn:Disconnect() fpsBoostConn = nil end

    if on then
        fpsBoostSaved = {
            globalShadows = Lighting.GlobalShadows,
            brightness = Lighting.Brightness,
            technology = nil,
        }
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
        end)
        pcall(function()
            UserSettings():GetService("UserGameSettings").SavedQualityLevel = Enum.SavedQualitySetting.QualityLevel1
        end)
        Lighting.GlobalShadows = false
        pcall(function()
            Lighting.EnvironmentDiffuseScale = 0
            Lighting.EnvironmentSpecularScale = 0
        end)
        -- su dalgaları / terrain
        pcall(function()
            if workspace.Terrain then
                workspace.Terrain.WaterWaveSize = 0
                workspace.Terrain.WaterWaveSpeed = 0
                workspace.Terrain.WaterReflectance = 0
                workspace.Terrain.WaterTransparency = 1
            end
        end)
        cleanVisualEffects(workspace)
        cleanVisualEffects(Lighting)

        fpsBoostConn = RunService.Heartbeat:Connect(function()
            -- ownership al (yakın partlar)
            local char = localPlayer.Character
            local root = char and (char:FindFirstChild("HumanoidRootPart") or char.PrimaryPart)
            if root then
                for _, part in ipairs(workspace:GetPartBoundsInRadius(root.Position, 80)) do
                    if part:IsA("BasePart") and not part.Anchored then
                        pcall(function()
                            if part:GetNetworkOwner() ~= localPlayer then
                                part:SetNetworkOwner(localPlayer)
                            end
                        end)
                        pcall(function()
                            sethiddenproperty(part, "NetworkIsSleeping", false)
                        end)
                    end
                end
            end
            -- periyodik temizlik (seyrek)
            if tick() % 2 < 0.05 then
                cleanVisualEffects(workspace)
            end
            Lighting.GlobalShadows = false
            pcall(function()
                if workspace.Terrain then
                    workspace.Terrain.WaterWaveSize = 0
                    workspace.Terrain.WaterWaveSpeed = 0
                end
            end)
        end)
    else
        if fpsBoostSaved then
            Lighting.GlobalShadows = fpsBoostSaved.globalShadows
            fpsBoostSaved = nil
        end
        pcall(function()
            settings().Rendering.QualityLevel = Enum.QualityLevel.Automatic
        end)
    end
end

FpsBoostToggle.MouseButton1Click:Connect(function()

end)

local function destroyNamedFolders()
    local targets = { "Events", "Light", "models", "Models", "lights", "Lights" }
    local function scan(parent)
        for _, child in ipairs(parent:GetChildren()) do
            local n = child.Name
            for _, t in ipairs(targets) do
                if n == t or n:lower() == t:lower() then
                    pcall(function() child:Destroy() end)
                    break
                end
            end
        end
    end
    pcall(function() scan(workspace) end)
    pcall(function() scan(game:GetService("ReplicatedStorage")) end)
    pcall(function()
        local rs = game:GetService("ReplicatedStorage")
        for _, child in ipairs(rs:GetDescendants()) do
            if child:IsA("Folder") or child:IsA("Model") then
                local n = child.Name:lower()
                if n == "events" or n == "light" or n == "models" or n == "lights" then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end)
    pcall(function()
        for _, child in ipairs(workspace:GetDescendants()) do
            if child:IsA("Folder") then
                local n = child.Name:lower()
                if n == "events" or n == "light" or n == "models" or n == "lights" then
                    pcall(function() child:Destroy() end)
                end
            end
        end
    end)
end

local function clearEspForChar(char)
    local data = espObjects[char]
    if not data then return end
    if data.highlight then data.highlight:Destroy() end
    if data.box then data.box:Destroy() end
    if data.beam then data.beam:Destroy() end
    if data.lineAtt0 then data.lineAtt0:Destroy() end
    if data.lineAtt1 then data.lineAtt1:Destroy() end
    if data.boxFolder then data.boxFolder:Destroy() end
    if data.speedGui then data.speedGui:Destroy() end
    if data.nameGui then data.nameGui:Destroy() end
    espObjects[char] = nil
end

local function getCharRoot(char)
    return char:FindFirstChild("HumanoidRootPart") or char:FindFirstChild("Torso") or char:FindFirstChild("UpperTorso")
end

local function updateAllEsp()
    local myChar = localPlayer.Character
    local now = tick()

    for _, plr in ipairs(Players:GetPlayers()) do
        local char = plr.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        local isLocal = (plr == localPlayer)

        if char and hum and hum.Health > 0 then
            local data = ensureEspForChar(char)
            local root = getCharRoot(char)

            -- Body / Box / Line sadece diğerleri
            if not isLocal then
                if data.highlight then
                    data.highlight.Enabled = bodyEspEnabled
                    data.highlight.FillColor = espBodyColor
                    data.highlight.OutlineColor = espBodyColor
                    data.highlight.OutlineTransparency = math.clamp(0.35 - (espBodyThickness - 1) * 0.1, 0.05, 0.5)
                    data.highlight.FillTransparency = math.clamp(0.65 - (espBodyThickness - 1) * 0.1, 0.35, 0.75)
                end

                if data.box then
                    data.box.Visible = boxEspEnabled
                    data.box.Color3 = espBoxColor
                    data.box.SurfaceColor3 = espBoxColor
                    data.box.LineThickness = 0.025 * espBoxScale
                end

                if data.beam then
                    data.beam.Enabled = lineEspEnabled
                    data.beam.Color = ColorSequence.new(espLineColor)
                    local w = 0.06 * espLineScale
                    data.beam.Width0 = w
                    data.beam.Width1 = w
                    local localRoot = myChar and getCharRoot(myChar)
                    if localRoot and data.lineAtt0 and data.lineAtt0.Parent ~= localRoot then
                        data.lineAtt0.Parent = localRoot
                        data.beam.Parent = localRoot
                    end
                    if root and data.lineAtt1 and data.lineAtt1.Parent ~= root then
                        data.lineAtt1.Parent = root
                    end
                elseif lineEspEnabled then
                    clearEspForChar(char)
                    ensureEspForChar(char)
                end

                -- Name ESP (ben hariç) + Roblox isim göstergesini kapat
                if data.nameGui and data.nameLbl then
                    data.nameGui.Enabled = nameEspEnabled
                    data.nameLbl.Text = plr.DisplayName ~= "" and plr.DisplayName or plr.Name
                    data.nameLbl.TextColor3 = Color3.fromRGB(240, 230, 255)
                end
                if hum then
                    if nameEspEnabled then
                        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None
                        hum.NameDisplayDistance = 0
                        pcall(function()
                            hum.HealthDisplayType = Enum.HumanoidHealthDisplayType.DisplayWhenDamaged
                        end)
                    else
                        hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.Viewer
                        hum.NameDisplayDistance = 100
                    end
                end
            else
                if data.highlight then data.highlight.Enabled = false end
                if data.box then data.box.Visible = false end
                if data.beam then data.beam.Enabled = false end
                if data.nameGui then data.nameGui.Enabled = false end
            end

            -- Speed ESP (herkes + ben)
            if data.speedGui and data.speedLbl and root then
                data.speedGui.Enabled = speedEspEnabled
                if speedEspEnabled then
                    local pos = root.Position
                    local speed = 0
                    if data.lastPos and data.lastTime then
                        local dt = now - data.lastTime
                        if dt > 0 then
                            speed = (pos - data.lastPos).Magnitude / dt
                        end
                    end
                    data.lastPos = pos
                    data.lastTime = now
                    data.speedLbl.Text = string.format("%.0f", speed)
                    data.speedLbl.TextColor3 = Color3.fromRGB(200, 120, 255)
                end
            end
        else
            if char then clearEspForChar(char) end
        end
    end

    for char, _ in pairs(espObjects) do
        if not char.Parent then
            clearEspForChar(char)
        end
    end
end

task.spawn(function()
    while task.wait(0.15) do
        pcall(updateAllEsp)
    end
end)

local function makeEdgePart(parent, size, cf, color)
    local p = Instance.new("Part")
    p.Name = "EspBoxEdge"
    p.Anchored = true
    p.CanCollide = false
    p.Material = Enum.Material.Neon
    p.Color = color
    p.Size = size
    p.CFrame = cf
    p.Parent = parent
    return p
end

local function buildBoxEdges(parent, center, size, thickness, color)
    local parts = {}
    local hx, hy, hz = size.X / 2, size.Y / 2, size.Z / 2
    local t = thickness
    -- 12 edges of a rectangular box
    local edges = {
        -- bottom
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, -hy, -hz))},
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, -hy, hz))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(-hx, -hy, 0))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(hx, -hy, 0))},
        -- top
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, hy, -hz))},
        {Vector3.new(size.X, t, t), CFrame.new(center + Vector3.new(0, hy, hz))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(-hx, hy, 0))},
        {Vector3.new(t, t, size.Z), CFrame.new(center + Vector3.new(hx, hy, 0))},
        -- vertical
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(-hx, 0, -hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(hx, 0, -hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(-hx, 0, hz))},
        {Vector3.new(t, size.Y, t), CFrame.new(center + Vector3.new(hx, 0, hz))},
    }
    for _, e in ipairs(edges) do
        table.insert(parts, makeEdgePart(parent, e[1], e[2], color))
    end
    return parts
end

local function refreshEspPreview()
    if espPreviewRefs.linePart then
        local s = 0.05 * espLineScale
        local z = espPreviewRefs.linePart.Size.Z
        espPreviewRefs.linePart.Size = Vector3.new(s, s, z)
        espPreviewRefs.linePart.Color = espLineColor
    end
    if espPreviewRefs.boxCenter and espPreviewRefs.boxSize then
        local world = EspViewport:FindFirstChildOfClass("WorldModel")
        if world then
            for _, p in ipairs(espPreviewRefs.boxParts) do
                if p and p.Parent then p:Destroy() end
            end
            local thickness = 0.035 * espBoxScale
            local pad = 0.15 * espBoxScale
            local sz = espPreviewRefs.boxSize + Vector3.new(pad, pad, pad)
            espPreviewRefs.boxParts = buildBoxEdges(world, espPreviewRefs.boxCenter, sz, thickness, espBoxColor)
        end
    end
    for _, p in ipairs(espPreviewRefs.bodyParts) do
        if p and p.Parent then
            local base = p:GetAttribute("BaseSize")
            if typeof(base) == "Vector3" then
                local expand = 0.12 * espBodyThickness
                p.Size = base + Vector3.new(expand, expand, expand)
            end
            p.Color = espBodyColor
            p.Transparency = math.clamp(0.4 - (espBodyThickness - 1) * 0.08, 0.2, 0.5)
        end
    end
end

local function buildEspPreview()
    local world = EspViewport:FindFirstChildOfClass("WorldModel")
    if world then world:Destroy() end
    world = Instance.new("WorldModel")
    world.Parent = EspViewport

    local cam = EspViewport:FindFirstChildOfClass("Camera")
    if not cam then
        cam = Instance.new("Camera")
        cam.Parent = EspViewport
    end
    EspViewport.CurrentCamera = cam

    -- Player: sol alt, sadece gövde/kollar/kafa görünsün, sağa baksın
    local playerModel
    local ok, result = pcall(function()
        return Players:CreateHumanoidModelFromUserId(localPlayer.UserId)
    end)
    if ok and result then
        playerModel = result
        for _, d in ipairs(playerModel:GetDescendants()) do
            if d:IsA("Script") or d:IsA("LocalScript") then d:Destroy() end
        end
        local hum = playerModel:FindFirstChildOfClass("Humanoid")
        if hum then hum.DisplayDistanceType = Enum.HumanoidDisplayDistanceType.None end
    else
        playerModel = createWhiteDummy()
    end
    playerModel.Name = "EspPlayer"
    playerModel.Parent = world
    -- kameraya bakan, hafif sağa dönük (ters değil)
    local pPos = Vector3.new(-2.1, -0.7, 1.0)
    playerModel:PivotTo(CFrame.new(pPos) * CFrame.Angles(0, math.rad(-15), 0))

    -- Bacon: en sağ-üst, oyuncunun yüzüne baksın
    local bacon = createBaconModel()
    bacon.Parent = world
    local bPos = Vector3.new(2.55, 2.0, -0.5)
    bacon:PivotTo(CFrame.lookAt(bPos, pPos + Vector3.new(0, 1.2, 0)))

    local playerRoot = playerModel.PrimaryPart or playerModel:FindFirstChild("HumanoidRootPart") or playerModel:FindFirstChild("Torso") or playerModel:FindFirstChildWhichIsA("BasePart")
    local baconRoot = bacon.PrimaryPart or bacon:FindFirstChild("Torso") or bacon:FindFirstChildWhichIsA("BasePart")

    -- Line ESP
    local linePart = Instance.new("Part")
    linePart.Name = "EspLine"
    linePart.Anchored = true
    linePart.CanCollide = false
    linePart.Material = Enum.Material.Neon
    linePart.Color = espLineColor
    linePart.Parent = world
    if playerRoot and baconRoot then
        local a = playerRoot.Position + Vector3.new(0, 1.6, 0)
        local b = baconRoot.Position + Vector3.new(0, 1.3, 0)
        local mid = (a + b) / 2
        local dist = (a - b).Magnitude
        linePart.Size = Vector3.new(0.05 * espLineScale, 0.05 * espLineScale, dist)
        linePart.CFrame = CFrame.lookAt(mid, b)
    end

    -- Box ESP: içi boş dikdörtgen (12 kenar)
    local cf, size = bacon:GetBoundingBox()
    local boxCenter = cf.Position
    local boxSize = size
    local thickness = 0.035 * espBoxScale
    local pad = 0.15 * espBoxScale
    local boxParts = buildBoxEdges(world, boxCenter, boxSize + Vector3.new(pad, pad, pad), thickness, espBoxColor)

    -- Body ESP: karakter dışını saran belirgin neon/kabuk
    local bodyParts = {}
    for _, bp in ipairs(bacon:GetDescendants()) do
        if bp:IsA("BasePart") and bp.Name ~= "Hair" then
            local shell = Instance.new("Part")
            shell.Name = "BodyEspShell"
            shell.Anchored = true
            shell.CanCollide = false
            shell.Material = Enum.Material.Neon
            shell.Color = espBodyColor
            shell.Transparency = 0.35
            local expand = 0.12 * espBodyThickness
            shell.Size = bp.Size + Vector3.new(expand, expand, expand)
            shell.CFrame = bp.CFrame
            shell:SetAttribute("BaseSize", bp.Size)
            shell.Parent = world
            table.insert(bodyParts, shell)
        end
    end

    espPreviewRefs.linePart = linePart
    espPreviewRefs.boxParts = boxParts
    espPreviewRefs.bodyParts = bodyParts
    espPreviewRefs.baconModel = bacon
    espPreviewRefs.playerModel = playerModel
    espPreviewRefs.boxCenter = boxCenter
    espPreviewRefs.boxSize = boxSize

    -- Kamera: sol-alt oyuncu + sağ-üst bacon
    cam.CFrame = CFrame.new(Vector3.new(0.1, 1.6, 9.2), Vector3.new(0.15, 0.85, 0))
end

local function makeSlider(order, minV, maxV, defaultV, onChanged)
    local holder = Instance.new("Frame")
    holder.Size = UDim2.new(1, -8, 0, 28)
    holder.BackgroundTransparency = 1
    holder.ZIndex = 213
    holder.LayoutOrder = order
    holder.Parent = EspSettingsRight

    local valueLbl = Instance.new("TextLabel")
    valueLbl.Size = UDim2.new(0, 36, 0, 14)
    valueLbl.Position = UDim2.new(1, -36, 0, 0)
    valueLbl.BackgroundTransparency = 1
    valueLbl.Text = string.format("%.1f", defaultV)
    valueLbl.TextColor3 = Color3.fromRGB(200, 180, 255)
    valueLbl.Font = Enum.Font.GothamBold
    valueLbl.TextSize = 10
    valueLbl.ZIndex = 214
    valueLbl.Parent = holder

    local track = Instance.new("Frame")
    track.Size = UDim2.new(1, -44, 0, 6)
    track.Position = UDim2.new(0, 4, 0, 16)
    track.BackgroundColor3 = Color3.fromRGB(35, 35, 45)
    track.ZIndex = 214
    track.Parent = holder

    local trackCorner = Instance.new("UICorner")
    trackCorner.CornerRadius = UDim.new(1, 0)
    trackCorner.Parent = track

    local fill = Instance.new("Frame")
    local ratio = (defaultV - minV) / (maxV - minV)
    fill.Size = UDim2.new(ratio, 0, 1, 0)
    fill.BackgroundColor3 = Color3.fromRGB(160, 30, 255)
    fill.ZIndex = 215
    fill.Parent = track

    local fillCorner = Instance.new("UICorner")
    fillCorner.CornerRadius = UDim.new(1, 0)
    fillCorner.Parent = fill

    local knob = Instance.new("Frame")
    knob.Size = UDim2.new(0, 12, 0, 12)
    knob.Position = UDim2.new(ratio, -6, 0.5, -6)
    knob.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    knob.ZIndex = 216
    knob.Parent = track

    local knobCorner = Instance.new("UICorner")
    knobCorner.CornerRadius = UDim.new(1, 0)
    knobCorner.Parent = knob

    local dragging = false
    local function setFromX(x)
        local absPos = track.AbsolutePosition.X
        local absSize = track.AbsoluteSize.X
        local r = math.clamp((x - absPos) / absSize, 0, 1)
        local val = minV + r * (maxV - minV)
        fill.Size = UDim2.new(r, 0, 1, 0)
        knob.Position = UDim2.new(r, -6, 0.5, -6)
        valueLbl.Text = string.format("%.1f", val)
        onChanged(val)
    end

    track.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            setFromX(input.Position.X)
        end
    end)
    knob.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
            setFromX(input.Position.X)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)

    return holder
end

local function loadAssetImage(targetImageLabel, id)
    targetImageLabel.Image = "rbxassetid://" .. tostring(id)
    local InsertService = game:GetService("InsertService")
    
    pcall(function()
        local result = InsertService:LoadAsset(id)
        if result then
            local decal = result:FindFirstChildOfClass("Decal") or result:FindFirstChildWhichIsA("Decal", true)
            if decal and decal.Texture then
                targetImageLabel.Image = decal.Texture
                return
            end
        end
    end)

    pcall(function()
        local rawData = game:HttpGet("https://assetdelivery.roblox.com/v1/asset/?id=" .. tostring(id))
        if rawData then
            local realId = rawData:match('<url>http[s]?://.-id=(%d+)</url>') or rawData:match('id=(%d+)')
            if realId then
                targetImageLabel.Image = "rbxassetid://" .. realId
            end
        end
    end)
end

        StealLabel_Anomaly.Text = "Stealing: None"
        updateStatusText_Anomaly(0)
        activePrompt_Anomaly = nil
    end
end)

local function finishIntro()
    if isIntroEnded then return end
    isIntroEnded = true

    if IntroSound then
        IntroSound:Stop()
        IntroSound:Destroy()
    end

    TweenService:Create(IntroFrame, TweenInfo.new(0.6), {BackgroundTransparency = 1}):Play()
    TweenService:Create(SkipButton, TweenInfo.new(0.3), {BackgroundTransparency = 1, TextTransparency = 1}):Play()

    for _, item in ipairs(topLetterLabels) do
        TweenService:Create(item.label, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
        TweenService:Create(item.stroke, TweenInfo.new(0.4), {Transparency = 1}):Play()
    end
    for _, item in ipairs(bottomLetterLabels) do
        TweenService:Create(item.label, TweenInfo.new(0.4), {TextTransparency = 1}):Play()
        TweenService:Create(item.stroke, TweenInfo.new(0.4), {Transparency = 1}):Play()
    end

    task.wait(0.6)
    IntroFrame:Destroy()

    OuterFrame_View.Visible = true
    MainFrame_Anomaly.Visible = true
    resetInactivity_View()
end

SkipButton.MouseButton1Click:Connect(finishIntro)

task.spawn(function()
    local startTime = tick()
    local lastShock = 0
    
    while tick() - startTime < 15 and not isIntroEnded do
        local elapsed = tick() - startTime
        
        local loudness = IntroSound.PlaybackLoudness or 0
        local normalizedLoudness = math.clamp(loudness / 400, 0, 1.5)
        
        OuterRing.Rotation = (OuterRing.Rotation - (2 + normalizedLoudness * 4)) % 360
        InnerRing.Rotation = (InnerRing.Rotation + (3 + normalizedLoudness * 5)) % 360
        
        local ringSizeOffset = normalizedLoudness * 40
        OuterRing.Size = UDim2.new(0, 360 + ringSizeOffset, 0, 360 + ringSizeOffset)
        OuterRing.Position = UDim2.new(0.5, -(180 + ringSizeOffset / 2), 0.5, -(180 + ringSizeOffset / 2))
        
        VignetteStroke.Transparency = math.clamp(0.4 - (normalizedLoudness * 0.25), 0.05, 0.5)
        
        for _, cardData in ipairs(bgCards) do
            cardData.frame.Rotation = cardData.baseRot + math.sin(elapsed * 3) * (5 + normalizedLoudness * 5)
            local scaleBonus = normalizedLoudness * 12
            cardData.frame.Size = UDim2.new(0, cardData.baseSize.X.Offset + scaleBonus, 0, cardData.baseSize.Y.Offset + scaleBonus)
            cardData.stroke.Transparency = math.clamp(0.4 - (normalizedLoudness * 0.3), 0.1, 0.6)
        end

        if loudness > 280 and (tick() - lastShock) > 0.45 then
            lastShock = tick()
            triggerShockwave()
        end

        task.wait(0.02)
    end

    finishIntro()
end)
